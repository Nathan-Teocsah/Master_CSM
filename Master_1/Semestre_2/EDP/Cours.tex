	\section{Dynamique de $N$ particules en interaction \small(\textit{François Castella})}
	\subsection{Formulation}
	
	\fbox{\red !} $N$ particules identiques, décrites classiquement par les équations de Newton \fbox{\red !}\\
	On ne part pas des équations de Schrödinger. 
	
	Donnons nous la fonction $F(x)$ qui est la force d’interaction entre 2 particule, c'est-à-dire :
	
	\includegraphics[scale = .5]{photo1.png}
	
	\begin{ex}[Attraction gravitationnelle]
		$ $
		\begin{center}
			\includegraphics[scale = .5]{photo2.png}
		\end{center}
		
		\begin{minipage}{18cm}
			Force subie par la terre du fait du Soleil : $-G\dfrac{Mm}{\|x\\|^2}\dfrac{x}{\|x\|} = -\dfrac{GMm}{\|T-S\|^3}(T-S)$.\\
			
			Force subie par le Soleil du fait de la Terre : $-G\dfrac{Mm}{\|x\|^2}\dfrac{x}{\|x\|} = -\dfrac{GMm}{\|T-S\|^3}(T-S)$.
		\end{minipage}
		
		\includegraphics[scale = .5]{photo3.png}
	\end{ex}
	
	On a (Newton) :$\underbrace{\text{masse}}_{%
		\shortstack{\rouge{= 1, par soucis}\\\rouge{de normalisation}}%
	}$ $\times$ accélération = force pour toutes les particules
	
	particule 1 :  $\begin{cases}
		\frac{d^2x}{dt^2}(t) = T(x_1(t)-x_2(t)) + F(x_1(t)-x_3(t)) + \ldots + F(x_1(t) - x_N(t))\\
		\vdots \\
		\frac{d^2x_N}{dt^2}(t) = F(x_N(t)-x_1(t)) + F(x_N(t)-x_2(t)) + \ldots + F(x_N(t)-x_{N-1}(t)) 
	\end{cases}$
	
	\newpage
	
	Soit :\\
	$
	\begin{cases}
		\forall i = 1,...,N$, $\frac{d^2x_i(t)}{dt^2} = \displaystyle\sum_{\underset{j\not=i}{j=1}}^N F(x_i(t) -x_j(t))\\
		\underline{\textcircled{+} \text{ données initiales à } t = 0}\\
		x_1(0) = x_1^0,\ldots,x_N^0(0) = x_N^0(0) \\
		v_1(0) = v_1^0,\ldots,v_N^0(0) = v_N^0(0)
	\end{cases}
	$
	
	avec \begin{minipage}[t]{15cm}
		$x_1^0,\ldots, x_N^0,$ positions initiales, données, chacune dans $\R^3$\\
		$v_1^0,\ldots, v_N^0,$ vitesses initiales, données , chacune dans $\R^3$
	\end{minipage}
	
	$\longrightarrow$ EDO d'ordre 2, en dimension $3N$ , avec typiquement, $N \simeq$ nombre d'Avogadro $\simeq 10^{23}$.\\
	\textbf{Calcul effectif inatteignable}.
	
	\par Cauchy-Lipschitz affirme que ce système admet une solution unique : d'où, $x_1(t),v_1(t),$ etc. sont entièrement déterminés par les données initiales.\\
	\textbf{\underline{Question :}} Que se passe t'il lorsque $N \to +\infty$ ?
	
	\par \underline{On a ici 2 limites naturelles :}\\
	\hspace*{2cm}\begin{minipage}[t]{17.5cm}
		\begin{enumerate}
			\item limite faible couplage (weak coupling limit) : hard sphere (sphère dure)\\
			Cas où les valeurs physique font que $F$ est de l'ordre de $\dfrac{1}{N}$, dans les unités physiquement pertinentes.\\
			On part alors de $\dfrac{d^2x_i}{dt^2}(t) = \dfrac1N \displaystyle\sum_{j\not = i} F(x_i(t)-x_j(t))$ et on passe à la limite $N \to \infty$
			\item limite faible densité (low density limit) : \\
			exemple typique de boule de billard de rayon $\epsilon$.
			
			\includegraphics[scale = .5]{photo4.png}				
			
			On assure $N\epsilon^2 = 1$ ($N\epsilon^2 =$ sommes de valeurs explorés par chaque particule), ceci assure (à vue de nez au moins) que en temps 1 on a de l'ordre 1 collision.\\
			$\rightarrow$ Dans ce régime de convergence, on a convergence, lorsque $N \to +\infty$ vers l'équation de Boltzmann.
		\end{enumerate}
	\end{minipage}
	
	\subsection{Limite Faible Couplage}
	
	\vspace{-.2cm}
	
	On part de Newton.\\
	$\dfrac{d^2 x_i}{dt^2}(t) = \displaystyle\sum_{j=1}^F(x_i(t)-x_j(t))$ pour $i = 1,\ldots,N$ et $F$ connue, avec par exemple : $F(x) = -Gm^2\dfrac{x}{\|x\|^3}$,\\
	avec \begin{minipage}[t]{4cm}
		$x_i(0) = x_i^0$ connue.\\
		$v_i(0) = v_i^0$ connue.
	\end{minipage}
	\begin{minipage}[t]{10cm}
		\textbf{Soit $T$ l'ordre de grandeur de l'échelle temporelle}
	\end{minipage}
	
	\textbf{$X$ l'ordre de grandeur de l'échelle spatiale} (si le travail a été bien fait).\\
	les $v_i^0$ et les $v_i(t)$ vont demeurer de l'ordre de $\frac{X}{T}$.\\
	\textbf{Soit $\m F$ l'ordre de grandeur} de $F$.
	
	\underline{Introduisons la notation suivante :}\\
	\hspace{3cm} $X_N(t) = (x_1(t),\ldots,x_N(t)) \in \R^{3\times N}, X_N^0 = (x_1^0,\ldots,x_N^0)$.\\
	$V_N(t) = (v_1(t),,\ldots,v_N(t)) \in \R^{3\times N}, V_N^0 = (v_1^0,\ldots,v_N^0)$.\\
	$z_i(t) = (x_i(t),v_i(t)) \in \R^6$ (variable de l'espace des phases).\\
	$z_i^0 = (x_i^0,v_i^0)$.\\
	$Z_N(t) = (z_1(t),\ldots,z_N(t))\in\R^{6\times N}, Z_N^0 = (z_1^0,\ldots,z_N^0)$\\
	Avec ces notations, les équations de Newton s'écrivent :
	\begin{center}
		
		$\begin{cases}
			m \dfrac{d^2 X_N}{dt^2}(t) = f(X_N(t)) \text{ où } f : \alpha \mapsto \left(\displaystyle\sum_{\underset{j\not = 1}{j=1}}^N F(\alpha_1-\alpha_j),\ldots,\sum_{\underset{j\not = N}{j=1}}^N F(\alpha_N-\alpha_j)\right)\\
			X_N(0) = X_N^0, V_N(0) = V_N^0
		\end{cases}$
	\end{center}
	
	\underline{Adimensionnons ce système :} \begin{minipage}[t]{15cm}
		Posons $t = T \overset{\sim}{t}$ \hspace{1cm} \begin{minipage}[t]{4 cm}
			où $\overset{\sim}{t}$  est d'ordre $1$ et $\overset{\sim}{t}$  est sans dimension.
		\end{minipage}\\		
		$x_i(t) = X_i(\overset{\sim}{t}) = X_i\left(\frac{t}{T}\right)$
	\end{minipage}
	
	De manière équivalente, on pose : $x_n(T) := x\overset{\sim}{X}_N\left(\frac{t}{T}\right)$.
	
	$\overset{\sim}{X}_N(\m F) = \dfrac{X_N(T\m F)}{X}$
	
	Soit $\overset{\sim}{V}_N(\m F) := \dfrac{d \overset{\sim}{X}_N(t)}{d\overset{\sim}{t}} = \dfrac{T}{X} V_N(T\m F)$
	
	\par Enfin, pour ce qui concerne le champ de force, on définit : $\begin{aligned}[t]
		\overset{\sim}{f}(\overset{\sim}{\alpha}) 
		&= \dfrac{f(X\overset{\sim}{\alpha})}{\m F} \\
		&= \left(\displaystyle\sum_{\underset{i\not=1}{i=1}}^N \dfrac{F(X(\overset{\sim}{\alpha}_i-\overset{\sim}{\alpha}_j))}{\m F}, \cdots \right)
	\end{aligned}$
	
	Ainsi partant de :
	\[
	m\dfrac{d^2X_N(t)}{dt^2} = f(X_N(t))
	\]
	
	On obtient, pour $\overset{\sim}{X}(\overset{\sim}{t}) = \dfrac{X_N(T\m F)}{X}$, la relation :
	\[
	\begin{aligned}[t]
		\dfrac{d^2\overset{\sim}{X}_N(t)}{d\overset{\sim}{t}^2} 
		&= \dfrac{T^2}{X}\left(T\m F\right) \\
		&= \dfrac{T^2}{X}\frac{1}{m}f(X_N(T\m F)) \\
		&= \dfrac{T^2\m F}{Xm} \overset{\sim}{f}(\overset{\sim}{X}_N(\m F))
	\end{aligned}
	\]
	
	c'est-à-dire : $\dfrac{d^2\overset{\sim}{X}_N(\overset{\sim}{t})}{d\overset{\sim}{t}^2} = \dfrac{T^2\m F}{X_m} \overset{\sim}{f}(\overset{\sim}X_N)$
	
	\fbox{+} $\overset{\sim}X_N(0) = \overset{\sim}X_N^0$, $\overset{\sim}{V}_N(0) = \overset{\sim}V_N^0$ \hspace{1cm} \fbox{+} $\overset{\sim}f(\overset{\sim}\alpha) = \left(\displaystyle\sum_{\underset{j\not = 1}{j=1}} \frac{F(X(\overset{\sim}\alpha_1-\overset{\sim}\alpha_j))}{\m F}, \cdots\right)$
	
	\fbox{Nous faisons désormais l'hypothèse que $\dfrac{T^2\m F}{X_m}$ est de l'ordre de $\dfrac{1}{N}$.}
	
	\par Partons des équations de Newton adimensionnées (en laissant tomber les tildes).
	\[
	\dfrac{d^2X_N}{dt^2}(t) = \dfrac{1}{N}G(X_N(t))
	\]
	où $G : \alpha \in \R^{3 \times N} \mapsto G(\alpha) = \left(\displaystyle\sum_{\underset{j\not = 1}{j=1}}^N F(x_1-x_j),\ldots,\sum_{\underset{j\not = N}{j=1}}^N F(\alpha_N-\alpha_j)\right)$\\ 
	et $F : \R^3 \to \R^3$ de classe $\m C^k$ avec $k \geq 1$.
	
	\par On souhaite procéder à la limite $N \to \infty$ dans ces équations (EDO en dimension $3\times $ (gros problème topologique, non linéaires)).
	
	\subsubsection{Fonction de répartition}
	
	\enc{red}{
		\underline{\textbf{Idée profonde :}} au lieu de se donner un jeu de données inutiles dons notre équation de Newton, on se donne une fonction $f^0(x_1^0,v_1^0,x_2^0,v_2^0,\ldots,x_N^0,v_N^0)$, dite fonction de répartition (ou fonction de densité) qui permet de dire :
		
		\[
		\mathbb{P}\left(x_1^0\in \Omega_1,v_1^0 \in O_1,\ldots,x_N^0 \in \Omega_N,v_N^0\in O_n \right) = \displaystyle\underset{\Omega_1\times O_1\times\cdots\times\Omega_N\times O_N}{\int}f^0(x_1^0,v_1^0,\ldots,x^0_N,v^0_N) dx_1^0dv_1^0\cdots dx_N^0dv_N^0
		\]
	}
	
	\fbox{\begin{minipage}[t]{17cm}
			\underline{Affirmation :} 
			
			\enc[17cm]{blue}{
					La fonction de répartition $f(t,x_1,t_1,\ldots,x_N,t_N)$ et le flot des équations de Newton : $\left(X_1(t,t_0,x^0_1,\ldots,v^0_N),V_1(t,t_0,x^0_1,\ldots,v^0_N),\ldots,X_N(t,t_0,x^0_1,\ldots,v^0_N),V_N(t,t_0,x^0_1,\ldots,v^0_N)\right)	$\\
					Vérifie :
					\begin{center}
						$f\left(t,X_1(t,t_0,x^0_1,\ldots,v^0_N),V_1(t,t_0,x^0_1,\ldots,v^0_N),\ldots,X_N(t,t_0,x^0_1,\ldots,v^0_N),V_N(t,t_0,x^0_1,\ldots,v^0_N)\right)$\\
						$=$\\
						$f^0(x^0_1,v^0_1,\ldots,x^0_N,v^0_N)$
					\end{center}}
	\end{minipage}}
	
	\underline{\textbf{Notation :}} \begin{minipage}[t]{15cm}
		$N$ particules \begin{minipage}[t]{15cm}
			\begin{center}
				$\left(\overbrace{X_1(t,0,\ldots)}^{flot},V_1(t,0,\ldots),\ldots,X_n(t,0,\ldots),V_N(t,0,\overbrace{x_1^0,v_1^0}^{=z_1^0 \in \R^6},\ldots,\overbrace{x_N^0,v_N^0}^{=z_N^0 \in \R^6})\right)$\\
				$ = $\\
				$Z_N(t,0,Z_N^0)$
			\end{center}
			,où $Z_n^0 = \left(z_1^0,\ldots,z_N^0\right) \in \R^{6N}$.
		\end{minipage}
		
		\vspace{.5cm}
		
		Et $Z_N(t,0,Z_N^0)$ est le flot des équations de Newtons que nous considérons.
	\end{minipage}
	
	\underline{\textbf{Reformulation :}} \begin{minipage}[t]{15cm}
		Nous avions introduit la répartition initiales des particules :
		\[f^0(x_1^0,v_1^0,\ldots,x_N^0,v_N^0) = f^0(Z_N^0)\]
		et la répartition à l'instant $t$ associée $f(t,x_1,v_1,\ldots,x_N,v_N) = f(t,Z_N^0)$\\
		
		\underline{Affirmation :} \enc[10cm]{blue}{\begin{center}
				$f(t,Z_N(t,0,Z_N^0)) = f^0(Z_N^0)$\\
				
				\vspace{.5cm}
				
				\includegraphics[scale=.5]{photo5.png}
		\end{center}}\\
		
		En effet : $Z_N^0 \in \Omega \iff Z_N(t,0,Z_N^0) \in Z_N(t,0,\Omega)$ car le flot :\\
		est, par exemple, $f^0(\alpha) = \mathbb{1}_\Omega(\alpha)$ pour un certain ouvert $\Omega \subset \R^{6N}$. 
		\begin{center}
			$\alpha\mapsto Z_N(t,0,\alpha)$\\
			est une bijection
		\end{center}
		
		\vspace{-.4cm}
		
		et dire que le système, à l'instant $t$, est dans $\Omega_t$ est équivalent à dire que le système est dans $\Omega$ à l'instant $0$.\\
		
		\begin{minipage}{8.2cm}
			\includegraphics[scale=.2]{photo6.png}
		\end{minipage} \begin{minipage}{10cm}
			Soit $f^0(\alpha) = 1 \iff f(t,Z_N(t,0,\alpha)) = 1$.\\
			c'est-à-dire $\forall \alpha \in \R^{6N}$\\
			$f(t,Z_N(t,0,\alpha)) = f^0(\alpha)$
		\end{minipage}
	\end{minipage}
	
	\vspace{.2cm}
	
	\textbf{\underline{NB :}} mathématiquement à l'horizon : $ f^0(\alpha), f(t,\alpha) \in L^1(\R^{6N},l\alpha) \text{ ou } \in L^1\cap L^\infty$
	
	\vspace{-.05cm}
	
	\textbf{\underline{Petit Calcul :}} \begin{minipage}[t]{16cm}
		Soit l'EDO $x'(t) = g(x(t))$ où $g$ est donnée, par exemple, par : $ g(\alpha) = \alpha(1-\alpha)$; $\alpha\in\R$.\\
		$\rightarrow$ Son flot est noté $X(t,0,x_0)$. On se donne une répartition : $f^0(x_0)$
		
		\begin{center}
			\includegraphics[scale=.25]{photo7.png}
		\end{center}
		
		
		et on se donne une fonction $f(t,x)$ qui vérifie $f(t,X(t,0,x_0)) = f^0(x_0)$\\
		\hspace*{-.5cm} Or : $\begin{aligned}[t]
			\dfrac{d}{dt}f(t,X(t,0,x_0)) 
			&\overset{\text{chain rule}}{=} 1\left(\dfrac{\partial f}{\partial t}\right)(t,X(t,0,x_0))+\underbrace{\left(\dfrac{dX(t,0,x_0)}{dt}\right)}_{\rouge{=g(X(t,0,x_0))}}\cdot\left(\dfrac{\partial f}{\partial x}\right)(t,,X(t,0,x_0))\\
			&\overset{EDO}{=} \left(\dfrac{\partial f}{\partial t}\right)(t, X(t,0,x_0)) + g(X(t,0,x_0))\cdot\left(\dfrac{\partial f}{\partial x}\right)(t,X(t,0,x_0))\\
			&\overset{rappel}{=} 0
		\end{aligned}$
	\end{minipage}
	
	\hspace{-.8cm}\enc{red}{\textbf{\underline{Conclusion :}} \begin{minipage}[t]{15cm}
			$\forall t, x$
			\[
			\dfrac{\partial f}{\partial t}(t,x) + g(x)\dfrac{\partial f}{\partial x}(t,x) 
			= \left(\dfrac{\partial f}{\partial t} + g\dfrac{\partial f}{\partial x}\right)(t,\underbrace{X(t,0,\overbrace{X(0,t,x)}^{\overset{(det)}{=x_0}})}_{\text{(rel de groupe pour le flot)} })=0
			\]
	\end{minipage}}
	
	\hspace{-.8cm}\enc{red}{\textbf{\underline{Conclusion :}} \begin{minipage}[t]{15cm}
			l'EDO $x'(t) = g(x(t))$ se traduit, pour la fonction $f$, transposée par le flot grâce à :
			\[
			f(t,X(t,0,x_0)) = f^0(x_0) \text{ , par l'EDP}
			\]
			(Dites du transport)
			
			\[
			\begin{cases}
				\dfrac{\partial f}{\partial t} (t,x) + g(x) \dfrac{\partial f}{\partial x}(t,x) = 0\\
				f(t=0,x) = f^0(x)
			\end{cases}
			\]
	\end{minipage}}

	Extension de ce propos en dimension supérieure.\\
	Soit l'EDO dans $\R^d$ : $\begin{cases}
		x'(t) = g(t,x(t)) \\
		x(0) = x_0 \in \R^d
	\end{cases}$, où $g : \begin{aligned}[t]
		\R \times \R^d
		& \to 
		&\R^d\\
		(t,\alpha)
		& \mapsto
		& g(t,\alpha)
	\end{aligned}$ est donnée.
	\begin{ex}[Newton] On considère :
		
		\[
		\begin{cases}
			X'(t) = V(t)\\
			V'(t) = G(t,X(t))\\
			X(0) = x_0, V(0) = v_0
		\end{cases}
		\]
		
		,où $G : \begin{aligned}[t]
			\R \times \R^3
			& \to \R^3\\
			(t,\alpha) 
			&\mapsto G(t,\alpha)
		\end{aligned}$. Soit $f^0 :\R^d \to \R$ (ou $\R_ +$).
	\end{ex}
	
	On définit $f(t,x)$ à travers :
	\[
	(2) f(t,X(t,0,x)) = f^0(x)
	\]
	Dérivons (2) par rapport à $t$ :
	\[
	\begin{aligned}
		0 
		&= \dfrac{d}{dt}(f^0(x)) = \dfrac{d}{dt}(f(t,\underbrace{X(t,0,x)}_{ = (X_1(t,0,x),\ldots,X_d(t,0,x))\in \R^d}))\\
		&=1\left(\dfrac{\partial f}{\partial t}\right)(t,X(t,0,x)) + \dfrac{dX_1}{dt}\cdot \left(\dfrac{\partial f}{\partial x_1}\right)(t,X(t,0,x)) + \ldots + \dfrac{dX_d}{dt}(t,0,x)\cdot\left(\dfrac{\partial f}{\partial x_1}\right)(t,X(t,0,x))\\
		&=\left(\dfrac{\partial f}{\partial t}\right)(t,X(t,0,x)) + g_1(t,X(t,0,x))\cdot\left(\dfrac{\partial f}{\partial x_1}\right)(t,X(t,0,x))+\ldots+g_d(t,X(t,0,x))\left(\dfrac{\partial f}{\partial x_d}\right)(t,X(t,0,x))\\
		&=\left(\dfrac{\partial f}{\partial t}\right)(t,X(t,0,x)) + \displaystyle\underbrace{\sum_{i=1}^d g_i(t,X(t,0,x)) \cdot\left(\dfrac{\partial  f}{\partial x_i}\right)(t,X(t,0,x))}_{=g(t,X(t,0,x))\cdot\nabla_x f(t,X(,0,x(t,0,x)))}\\
		&= \dfrac{\partial f}{\partial t}(t,X(t,0,x)) + g(t,X(t,0,x))\cdot\left(\nabla_x f\right)(t,X(t,0,x))
	\end{aligned}
	\]
	
	\enc{red}{\textbf{\underline{Conclusion :}} \begin{minipage}[t]{11cm}
			$\forall t, \forall x$
			\[
			0 = \left(\dfrac{\partial f}{\partial t}(t,z) + g(t,z)\nabla_zf(t,z)\right)(t,z=X(t,0,x))
			\]
			Soit $(t,x)$ fixé. Posons : $z = X(t,0,x)$ ($\iff x = X(0,t,z)$)
	\end{minipage}}
	
	On a par (3) :
	\[
	\forall t,\forall z, \dfrac{\partial f}{\partial t}(t,z) + g(t,z) \cdot \left(\nabla_xf\right)(t,z) = 0
	\]
	
	Pour recoller tout cela nous avons montré :
	\begin{thm}[Méthode des caractéristiques]$ $\\
		Soit l'EDO $x'(t) = g(t,x(t))$ de flot $X(t,0,x)$.\\
		Soit $f(t,x)$ satisfaisant  : $f(t,X(t,0,x)) = f^0(X)$.\\
		On a :
		\[
		\begin{cases}
			\forall t, \forall x, \dfrac{\partial f}{\partial t}(t,x)+f(t,x)\cdot\nabla_x f(t,x) = 0\\
			\forall x, f(t=0,x) = f^0(x)
		\end{cases}
		\]
	\end{thm}
	
	
	\textbf{\underline{Dans le cas de Newton, tout cela donne : }}
	\[f^0(x,v) = f(t,X(t,0,v),V(t,0,x,v))\]
	
	$\implies \begin{aligned}[t]
		0
		&= \dfrac{d}{dt}(f^0(x,v))\\
		&= \left(\dfrac{\partial f}{\partial t}\right)(t,\ldots) + V(t,0,x,v)\cdot\left(\nabla_x f\right)(t,\ldots) + G(t,X(t,0,x,v))\cdot\left(\nabla_v f\right)(t,\ldots)
	\end{aligned}$\\
	
	A la fin on obtient :
	\[
	\forall t, \forall x, \forall v, \dfrac{\partial f}{\partial t}(t,x,v)+v\cdot\nabla_xf(t,x,v) + G(t,x)\cdot\nabla_v f(t,x,v) \text{ \hspace{.5cm} (4)}
	\]
	\rouge{Ici nous disons que l'EDP provient de l'EDO : $\begin{cases}
			X'(t) = V(t) \\
			V'(t) = G(t,X(t))
		\end{cases}$}
	
	avec $z = (x_1,v_1,\ldots,x_N,v_N)$.
	
	$X_i(t=0,0,z) = x_i$
	
	$V_i(t=0,0,z) = v_i$
	
	\begin{minipage}[t]{17cm}
		On a déjà vu que, si $f(t,x_1,v_1,\ldots,x_N,v_N)$ = \begin{minipage}[t]{10cm}
			la densité de particule à l'instant $t$ en $(x_1,v_1,\ldots,x_N,v_N)$
		\end{minipage}
	\end{minipage}
	
	qui $f(t,X_1(t,0,z),V_1(t,0,z),\ldots,X_d(t,0,z),V_d(t,0,z)) = f^0(z)$, où $f^0 = $ la densité de particules à l'instant 0. Nous en déduisons que $f(z)$ vérifie l'EDP de transport :
	
	$f(t=0,z) = f^0(z)$ (donnée de l'équation de transport)
	
	\[
	\dfrac{\partial f}{\partial t}(t,x_1,v_1,\ldots,v_N,v_N) + \displaystyle\sum_{k=1}^N v_k \cdot \nabla_{x_k} f + \sum_{k=1}^N \left[\frac{1}{N} \sum_{\underset{j\not=k}{j=1}}^N F(x_k-x_j)\right]\cdot\nabla_{v_k}f = 0
	\]
	
	$\rightarrow$ \begin{minipage}[t]{15cm}
		EDP linéaire d'inconnue $f(t,z)$.\\
		avec, typiquement, $f(t,z) \in C^1(t\in\R, L^1(\R^{6N}))$
	\end{minipage}
	
	\newpage
	
	\subsubsection{Marginale de $f$}
	
	\underline{\underline{\textbf{Idée :}}} introduire les marginales de $f$, définies par :
	
	\[
	f_N^1(t,x_1,v_1) = \displaystyle\int_{\R^6\times\R^6 \times \ldots \times \R^6} f(t,x_1,v_1,\ldots,x_N,v_N)dx_2dv_2\cdots dx_N dv_N 
	\]
	
	\[\vdots\]
	
	\[
	f_N^k(t,x_1,v_1,\ldots,x_k,v_k) = \displaystyle\int_{\R^{6(N-k)}} f(t,x_1,v_1,\ldots,x_N,v_N)dx_kdv_k\cdots dx_N dv_N
	\]
	
	\[\vdots\]
	
	Il est naturel d'affirmer que chaque fonction $f_N^K$ possède une limite (faible) $f^k$ lorsque $N \to \infty$.
	Pour calculer la limite de $f_N^k$ pour chaque $k = 1,\ldots,N$, écrivons l'EDP satisfaisant pour chacune de ces fonctions. Pour ce faire, supposons que pour toute \textit{permutation} $\sigma \in \m S_N$:
	
	\[
	f^0(x_1,v_1,\ldots,x_N,v_N) = f^0(x_{\sigma(1)},v_{\sigma(1)},\ldots,x_{\sigma(N)},v_{\sigma(N)}) \text{\hspace{.5cm} (1)}
	\]
	
	$\rightarrow$ les particules sont interchangeables.
	
	Les numéros des particules sont interchangeable initialement à $t = 0$.
	
	\[
	f^0(x_1,v_1,\ldots,x_N,v_N) = g^0(x_1,v_1)\cdots g^0(x_N,v_N) \text{\hspace{.5cm} (2)}
	\]
	
	pour une certaine fontion $g^0$ (on suppose que les particules sont initialement réparties indépendamment et selon la même loi).
	
	\begin{thm}[Admis à ce stade]
		\begin{minipage}[t]{10cm}
			Sous l'hypothèse de l'équation (1), $\forall t$:
		\end{minipage}
		
		\[
		f(t,x_1,v_1,\ldots,x_N,v_N) = f(t,x_{\sigma(1)},v_{\sigma(1)},\ldots,x_{\sigma(N)},v_{\sigma(N)})
		\]
		
		La propriété d'interchangeabilité est conservée avec le temps.
	\end{thm}
	
	Fort de cette information écrivons l'équation satisfaite par $f_N^1$, on part de :
	
	\[
	\underbrace{\dfrac{\partial f}{\partial t}(t,x_1,v_1,\ldots,v_N,v_N)}_{(1)} + \underbrace{\displaystyle\sum_{k=1}^N v_k \cdot \nabla_{x_k} f}_{(2)} + \underbrace{\sum_{k=1}^N \left[\frac{1}{N} \sum_{\underset{j\not=k}{j=1}}^N F(x_k-x_j)\right]\cdot\nabla_{v_k}f}_{(3)} = 0
	\]
	
	On observe que :
	
	\begin{enumerate}
		\item[$(1)$] $\begin{aligned}[t]
			\dfrac{\partial}{\partial t} f_N^1(t,x_1,v_1) 
			&= \dfrac{\partial}{\partial t} \int f(t,\ldots) dx_2dv_2 \cdots dx_N dv_N\\
			&= \int \dfrac{\partial f}{\partial t}(t,\ldots) dx_2 dv_2 \ldots dx_N dv_N
		\end{aligned}$
		
		\item[$(2)$] ($k=1$), $\displaystyle\int v_1 \cdot \nabla_{x_1} f(t,\ldots) dx_2 dv_2 \ldots dx_N dv_N = v_1 \nabla_{x_1}\left[\int f(t,\ldots)dx_1\ldots dv_N\right] = v_1 \cdot \nabla_{x_1} f_N^1$
		
		\hspace{-.6cm}$(k \not= 1)$, $\displaystyle\int v_k \cdot \nabla_{x_k} f(t,\ldots) dx_2dv_2 \ldots dx_N dv_N = -\int f(t,\ldots)\text{div}_{x_1}(v_k) dx_2 \cdots dv_N = 0$
		
		\rouge{\underline{Rappel :}} $\displaystyle \int_{\R^d} u(y) \cdot \nabla_y \phi(y) dy = - \int \phi(y)\text{div}_y(u(y)) dy$\\
		
		\item[$(3)$] $(k=1)$, $\begin{aligned}[t]
			&\displaystyle\int \frac{1}{N} \sum_{\underset{j\not=1}{j=1}}^N F(x_1-x_j)\cdot \nabla_{v_1} f dx_1 \cdots dv_N\\
			=
			&\frac{1}{N} \displaystyle\sum_{\underset{j\not=1}{j=1}}^N \int F(x_1-x_j) \cdot \nabla_{v_1} f(t,\ldots)\\
			=
			&\frac{1}{N}\displaystyle\sum_{\underset{j\not=1}{j=1}}^N\int F(x_1-x_j)\nabla_{v_1}f(t,x_1,v_1,x_{\rouge j},v_{\rouge j},\dots,x_{\rouge2},v_{\rouge2},\ldots,x_N,v_N) \\
			=
			&\frac{1}{N}\displaystyle\sum_{\underset{j\not= 1}{j=1}}^N\underbrace{\int F(x_1-x_j)\cdot \nabla_{v_1}f(t,x_1,v_1,\ldots)}_{\text{\rouge{pas d'indice j !}}}\\
			=
			& \frac{N-1}{N} \int F(x_1-x_2) \cdot \nabla_{v_1}f(t,\cdots) dx_2 \cdots dv_N\\
			=
			& \frac{N-1}{N} \int F(x_1-x_2)\nabla_{v_1} f(t,,\cdots) dx_3\cdots dv_N dx_2 dv_2\\
			=
			& \frac{N-1}{N} \int F(x_1-x_2) \cdot \nabla_{v_1} \int f_N^2(t,\ldots) dx_2 dv_2
		\end{aligned}$\\
		
		($k\not = 1$), $\begin{aligned}[t]
			&\displaystyle\frac{1}{N}  \sum_{\underset{j\not=k}{j=1}}^NF(x_k-x_j)\nabla_{v_k}f(t,x_1,\ldots,x_N)\\
			\underset{IPP}{=}
			& \frac{1}{N} \sum_{\underset{j\not= k}{j=1}}^N -\int \underbrace{\text{div}_{v_j}(F(x_k-x_j))}_{=0}.f(t,\cdots) dx_2 \cdots dv_N
		\end{aligned}$		
	\end{enumerate}
	
	C'est-à-dire, on a : $\dfrac{\partial}{\partial t} f_N^1(t,x_1,v_1) + v_1\cdot \nabla_{x_1} f_N^1 + \dfrac{N-1}{N} \displaystyle\int F(x_1-x_2)\cdot \nabla_{v_1} f_N^2(t,x_1,v_1,\ldots)dx_2 dv_2 = 0$.
	
	De même :
	
	$ \begin{aligned}[t]
		0 = \dfrac{\partial}{\partial t} f_N^2(tx_1,v_1,x_2,v_2)
		&+v_1 \cdot \nabla_{x_1}f_N^2 + v_2 \cdot\nabla_{x_2}f_N^2 \\
		&+ \dfrac{N-2}{N}\displaystyle\int F(x_1-x_4)\nabla_{v_1}f_N^3(\cdots)dx_4 dv_3 + \dfrac{N-2}{N}\displaystyle\int F(x_2-x_3)\nabla_{v_2}f_N^3(\cdots)dx_4 dv_3 \\
		&+ \frac{1}{N}F(x_1-x_2)\nabla_{v_2} f_N^2 + \frac{1}{N}F(x_2-x_1)\nabla_{v_1}f_N^2
	\end{aligned}$
	
	On récupère ainsi une famille d'équation de transport de la forme : $\begin{aligned}[t]
		\dfrac{\partial }{\partial t} f_N^1 
		&= \cdots f_N^2 \cdots\\
		\dfrac{\partial}{\partial t} f_N^2
		&= \cdots f_N^3 \cdots\\
		&\vdots\\
		\dfrac{\partial}{\partial t} f_N^k
		&= \cdots f_N^{k+1} \cdots
	\end{aligned}$

	\subsubsection{Passage à la limite}
	
	\hspace{-.8cm}\enc[19.5cm]{red}{On a vu que les marginales $f_N^k(t,x_1,v_1,\ldots,x_k,v_k)$ vérifient :
	
	$\dfrac{\partial f_N^1}{\partial t}(t,x_1,v_1) + v_1\cdot\nabla_{x_1}f_N^1 + \dfrac{N-1}{N} \displaystyle\int F(x_1-x_2)\nabla_{v_1} f_N^2(t,x_1,v_1,x_2,v_2) = 0$\\
	
	$\dfrac{\partial f_N^2}{\partial t} + v_1 \nabla_{x_2}f_N^2 + \dfrac{N-2}{N}\left[\displaystyle\int F(x_1-x_3)\nabla_{v_1} f_N^3 + F(x_2-x_3)\nabla_{v_2}f_N^3\right] + \dfrac{1}{N}\nabla_{v_1} f_N^2 + \dfrac{1}{N}F(x_2-x_1)\cdot\nabla_{v_2}f_N^2 = 0$}
	
	\underline{Quand $N \to \infty$ :} \begin{minipage}[t]{16cm}
		
		\vspace{-.25cm}
		
		\begin{enumerate}
		\item  Chaque suite de fonction $f_N^k$ possède naturellement une limite (\underline{\underline{faible}}, au sens des distributions), lorsque $N\to\infty$, notée $f^k$.
		
		\item  On saura donc passer à la limite(faible) dans les EDP linéaires satisfaites par les $f_N^k$.
	\end{enumerate}
	\end{minipage}
	
	
	
	\hspace{-.8cm}\enc[19.5cm]{red}{\underline{A la limite on a :}\\
		
		$(S)\begin{cases}
		\dfrac{\partial}{\partial t} f^1(t,x_1,v_1) + v_1\cdot\nabla_{x_1} f^1 + \displaystyle\int_{\R^6} F(x_1-x_2)\cdot\nabla_{v_1}f^2(t,x_1,v_1,x_2,v_2) dx_2 dv_2 = 0\\\\	
		\dfrac{\partial }{\partial t} f^2(t,x_1,v_1,x_2,v_2) + v_1 \nabla_{x_1}f^2 + v_2\nabla_{x_2}f^2 + \displaystyle\int\left[F(x_1-x_3)\cdot\nabla_{v_1}f^3 + F(x_2-x_3)\nabla_{v_2} f^3\right]dx_3v_3 = 0
	\end{cases}$}\\
	
	\rouge{\fbox{!}} le terme \rouge{d'autointeraction} disparaît \rouge{\fbox{!}}
	
	\vspace{.5cm}
		
	\hspace{-.8cm}Nous avons supposé $f^0(x,x_1,v_1,\cdots,x_N,v_N) = g^0(x_1,v_1)\times\cdots\times g^0(x_N,v_N)$ (indépendance des particules).
	
	D'où : \begin{minipage}[t]{18cm}
		$f^1(t=0,x_1,v_1) = g^0(x_1,v_1)$
		
		$f^2(t,x_1,v_1,x_2,v_2) = g^0(x_1,v_1)g^0(x_2,v_2)$
		
		etc.
	\end{minipage}
	
	\textbf{\underline{Si}} l'indépendance initiale des particules demeure vraie aux instants $t>0$ ultérieure, ainsi que l'hypothèse de distribution identique,\\
	\textbf{\underline{Alors}} $f^2(t,x_1,v_1,x_2,v_2) = f^1(t,x_1,v_1)\times f^1(t,x_2,v_2)$ et $\forall k, f^k(t,x_1,v_1,\cdots,x_k,v_k) = f^1(t,x_1,v_1) \times\cdots\times f^1(t,x_k,v_k)$.
	
	Ainsi nécessairement, l'eq sur $f^1$ devient : \\
	
	\enc{lightgray}{\[\dfrac{\partial f^1}{\partial t}(t,x_1,v_1) + v_1\nabla_{x_1}f^1 + \displaystyle\int_{\R^6} F(x_1-x_2)\nabla_{v_1}\left[f^1(t,x_1,v_1)f^1(t,x_2,v_2)\right]=0 \text{\hspace{1cm} (Vlasov)}\] }
	
	Ce qui nous donne une équation sur $f^1$.
	
	Ainsi prouver que l'équation de Vlasov sur $f^1$ nous donne une unique solution $f^1(t,x_1,v_1)$ (ok, admis), on connait$f^1$, mais $f^2,\cdots,f^k,\cdots$ : On connait la lmite des $f_N^k$, $\forall k$.
	
	\par Ayant ainsi définit $f^k$ comme $f^1(t,x_1,v_1)\times\cdots\times f^1(t,x_k,v_k)$ vérifiant, réciproquement, que ce choix de $f^k$ vérifie $(S)$.

	\subsection{Limite Faible Densité}

		\dfrac{dX_i}{dt} = v_i\\
		\dfrac{dV_i}{dt} = \dfrac{1}{N}\displaystyle\sum_{\underset{j\not=i}{j=1}}^N F(x_i-x_j)
	\end{cases}\text{ et } N\to\infty\]
	
	Maintenant on va passer à la limite suivant la même méthodologie sur la limite faible densité.\\
	
	\begin{minipage}{8cm}
	$\begin{cases}
		\dfrac{dX_i}{dt} = V_i\\
		\dfrac{dV_i}{dt} = \underbrace{ }_{\rouge{1}} \displaystyle\sum_{\underset{j\not=i}{j}}^N F\left(\dfrac{X_i-X_j}{\rouge{\fbox{$\epsilon$}}}\right)

	Dans le cas des \underline{sphères dures} (collisions élastiques, avec conservation de l'énergie au cour de chaque choc). L'équation sur $f$ est :
	
	\[
	\dfrac{\partial f}{\partial t}(t,x_1,v_1,\cdots,x_N,v_N) + \displaystyle\sum_{i=1}^N v_i\cdot \nabla_{x_i} f = \underbrace{\dfrac{1}{2}}_{\underset{\to \underline{Armand}}{\underset{\text{les collisions dans la somme}}{\text{On compte 2 fois}}}} \sum_{i,j=1}^N \underbrace{\mathcal L_{i,j}\left[f\right]}_{\rouge{\underset{\text{collision entre }i\text{ et }j}{\text{terme de}}}}
	\]
	
	où :
	\[
	\begin{aligned}
		\mathcal{L}_{i,j}\left[f\right](t,x_1,v_1,\cdots,x_N,v_N) 
		&= \displaystyle\int_{\mathbb{S}^2}\begin{aligned}[t]
			\Big[
			 &f(t,x_1,v_1,\cdots,x_i,v_i',\cdots,x_j=x_i+\varepsilon\omega,v_j',\cdots,x_N,v_N)\varepsilon^2\left<v_j-v_i|\omega\right>_+ \\

			\end{aligned}\\
		&=+\text{terme de gain}-\text{terme de perte}
	\end{aligned}
	\]
	
	\begin{minipage}{10cm}
		\includegraphics[scale=.6]{photo12.png}
	\end{minipage}\begin{minipage}{15cm}
	avec $\underset{\rouge{\text{conservation de l'énergie cinétique}}}{\begin{cases}
		v_i'+v_j'=v_i+v_j\\
		(v_i')^2+(v_j')^2 = v_i^2+v_j^2
	\end{cases}}$\\
	où $\begin{cases}
		\omega \in \R^3\\

	\begin{thm}[Pas facile, admis]

	\subsection{Entropie}
	
	On a vu la CV, dans la limite faible densité pour $N$ sphères dures en interactions, vers :
	
	\[
	\dfrac{\partial f}{\partial t}(t,x,v) + v\nabla_x f = Q(f,f)(t,x,v)
	\]
	
	avec $Q(f,f) = \displaystyle\int \left[f'f'_\star - ff_\star\right]\abs{\langle v-v_\star|\omega\rangle}d\omega dv_\star$.\\
	Où $\begin{cases}
		v' = v-\langle v-v_\star |\omega\rangle\omega \\
		v'_\star = v_\star + \langle v-v_\star |\omega\rangle\omega
	\end{cases}$.\\
	Calculer $\begin{aligned}[t]
		\dfrac d{dt} \underbrace{\left[\displaystyle\int f(t,x,v)\log(f(t,x,v))dxdv\right]}_{\rouge{\text{entropie de $f$}}} 
		&= \int (\partial_t f)\log(f) + f \dfrac{\partial_t f}{f}dxdv\\
		&= \int \left[\log f + 1\right]Q(f,f) = v\nabla_x f dxdv
	\end{aligned}$\\
	Or $\begin{aligned}[t]
	\displaystyle\int \left[\log f + 1\right]v\nabla_x f dx dv 
	&= \int \text{div}_x\left(v\log f(t)\right)f(t,x,v)dxdv \\
	&= - \int fv\cdot\nabla_x\left[\log f +1\right]dxdv \\
	&= - \int f\sum_i v_i \dfrac{\partial_{x_i} f}{f} dxdv = -\int v\cdot \nabla_x f dxdv\\
	&= + \int \underbrace{\left(\text{div}_x v\right)}_{=0} f dxdv = 0 
	\end{aligned}$
	\\
	De plus $I := \displaystyle\int \left(\log f +1\right)Q(f,f)dxdv = \int \left(\log f +1\right)\left[\underset{\rouge{\text{en $(x,v')$ }}}{f'}\underset{\rouge{\text{en $(x,v_\star')$}}}{f_\star'}-\underset{\rouge{\text{en $(x,v)$ }}}f\underset{\rouge{\text{en $(x,v_\star)$}}}{f_\star}\right]\abs{\langle v-v_\star|\omega\rangle}d\omega dx dv dv_\star$\\
	
	Où $\begin{cases}
		v' = v-\langle v-v_\star|\omega\rangle\omega\\
		v'_\star = v_\star + \langle v-v_\star|\omega\rangle\omega
	\end{cases}$, ou encore $\begin{pmatrix}
	v' \\ v_\star'
	\end{pmatrix} = T_\omega\begin{pmatrix}
	v \\ v_\star
	\end{pmatrix}$\\
	
	Donc $\begin{aligned}[t]
		I 
		&= \displaystyle\int \left(\log f + 1\right)\left[f'f_\star'-ff_\star\right] \abs{\langle v-v_\star|\omega\rangle} d\omega dx dv dv_\star\\
		&\underset{(v,v_\star)\to(v_\star,v)}= \int \left(\log f_\star\right)\left[f'f'_\star-ff_\star\right]\abs{\langle v-v_\star|\omega\rangle}d\omega dx dv dv_\star
	\end{aligned}$\\
	
	De même : $I =$\\
	\hspace*{-1cm}$\displaystyle\int \left[\left(\log f(t,x,v(v,v_\star',\omega))\right)+1\right]\left(f'f'_\star-f(t,x,v(v',v_\star',\omega))f(t,x,v_\star(v',v_\star',\omega))\right)\abs{\langle v(\cdots)-v_\star(\cdots)|\omega\rangle} \abs{\det T_\omega}dxd\omega'dv_x'd\omega$
	
	Donc : $\begin{aligned}[t]
		I 
		&= \displaystyle\int \left[\log f(t,x,v')+1\right]\left(ff_\star-f'f_\star'\right)\abs{\langle v'-v'_\star|\omega\rangle}d\omega dx dv dv_\star\\
		&=\int \left[\log f' + 1\right]\left(ff_\star-f'f_\star'\right)\abs{v-v_\star}d\omega dx dv dv_\star\\
		&= \displaystyle\int \left[\log f'_\star + 1\right]\left(ff_\star - f'f_\star'\right)\abs{\langle v-v_\star|\omega}d\omega dx dv dv_\star
	\end{aligned}$\\
	
	On a donc 4 expressions différentes pour I.En faisant la moyenne de ces 4 expressions, on a :
	\[
	\begin{aligned}
		I 
		&= \displaystyle\int \dfrac{\left(\log f +1\right)+\left(\log f_\star + 1\right) - \left(\log f' +1\right)-\left(\log f'_\star + 1\right)}{4}\left[f'f'_\star-ff_\star\right]\abs{\langle v-v_\star|\omega\rangle}\\
		&=  \frac14 \int\left(\log f + \log f_\star - \log f' - \log f'_\star\right)\abs{f'f'_\star-ff_\star}\abs{\langle v-v_\star|\omega\rangle}d\omega dx dv dv_\star \\

	\]

Soit $f(t,x,v)$ qui représente le nombre de particule qui sont en $x$ à l'instant $t$ avec la vitesse $v$.\\

On va considérer un système physique où aucune force n'agit sur les particules et il n'y a pas de collision. Par ailleurs, on considère  que l'espace est infini.\\

$f(t,x,v) = f(t=0,x-tv,v) = f_0(x-vt,v)$

\[
\begin{aligned}
	f(t,x,v) & = f(t=0,x-tv,v) \\
	&= f_0(x-vt) \hspace{1cm} \text{où $f_0$ dépend uniquement de $x$ et $v$}\\
	&=f_0(x_1-v_1t,x_2-v_2,x_3-v_3t,v_1,v_2,v_3) \\

	&= -v\cdot\nabla_x f_0(x-vt,v) \\
	\nabla_x f(t,x,v) &= \nabla_x f_0(x-vt,v)
\end{aligned}

\textbf{équation de Boltzmann}

$\partial_t f + v\nabla_x f = Q(f,f)$, \hspace{.2cm} où $Q(f,f) - \underbrace{Q^+(f,f)}_{gain}-\underbrace{Q^-(f,f)}_{perte}$

\[
Q^-(f,f) = \displaystyle\int_{\mathbb{S}^*}\int_{\R^3} f_\star(t,x,v)B(|\!|v-v_\star|\!|)\mathrm dv_\star d\sigma
\]

$\begin{aligned}
	Q^+(f,f) &= \displaystyle\int\int \underbrace{f(t,x,v')}_{f'}\underbrace{f(t,x,v_\star')}_{f_\star'}B(|\!|v-v_\star|\!|)\mathrm dv_\star\mathrm dv\\
	Q(f,f) &= \int\int \left(f'f_\star' - ff_\star\right)B(|\!|v-v_\star|\!|)
\end{aligned}$

Préserver masse, quantité de mouvement et énergie revient à avoir les égalités :

\[
\langle Q(f,f),\Psi\rangle = \displaystyle\int\int\int \left(f'f_\star'-ff_\star\right)B(|\!|v-v_\star|\!|)\Psi(v)\mathrm dv \mathrm dv_\star \mathrm d\sigma
\]

Masse $\Psi(v) = m_p$\\
Quantité de mouvement : $\Psi(v) = m_pv$\\
Énergie cinétique : $\Psi(v) = \frac12 m_p v^2$\\
$\begin{aligned}
	\langle Q(f,f),m_p\rangle &= 0\\
	\langle Q(f,f), m_p v\rangle &=0 \\
	\langle Q(f,f),m_pv^2\rangle &=0
\end{aligned}$	\\

On a la propriété que $\langle Q(f,f), \Psi\rangle = \frac12 \displaystyle\int\int ff^\star\left(\Psi_\star+\Psi_\star'-\Psi-\Psi^\star\right)B(|\!|v-v_\star|\!|)\mathrm dv \mathrm dv_\star \mathrm d\sigma$\\

$\begin{aligned}
	\langle Q(f,f),\Psi\rangle &= \displaystyle\int\int\int \left(f'f_\star'-ff_\star\right)\Psi B(|\!|v-v_\star|\!|)\mathrm dv \mathrm dv_\star \mathrm d\sigma\\
	&= \int\int\int \left(f'f_\star' - ff_\star\right)\Psi_\star B(|\!|v-v_\star|\!|)\mathrm dv \mathrm dv_\star \mathrm d\sigma \\
	&= \frac12 \int\int\int \left(f'f_\star'-ff_\star\right)\left(\Psi+\Psi^\star\right)\mathrm dv\mathrm dv_\star \mathrm d\sigma \\
	&= \frac12 \int\int \left(ff_\star - f'f_\star'\right)\left(\Psi'+\Psi_\star'\right)\mathrm dv\mathrm dv_\star\mathrm\sigma
\end{aligned}$

$\langle Q(f,f),\Psi\rangle = \frac14 \displaystyle\int\int\int \left(f'f_\star'-ff_\star\right)\left(\Psi_+\Psi_\star-\Psi_\star'-\Psi'\right)B(|\!|v-v_\star|\!|,\sigma)\mathrm dv \mathrm dv_\star \mathrm dv$\\
$\hookrightarrow$ donne les conservations\\
$H(f,f) = \displaystyle\int\left(f\ln f - f\right) \mathrm dv$ (entropie au sens mathématiques)\\
$H(f)$ ne dépend que de $f$ \hspace{.5cm} $\begin{aligned}[t]

\end{aligned}$\\

$\begin{aligned}

	&= \int \left[Q(f,f)-(v\cdot\nabla_x f)\right]\ln f \mathrm dv \\
	&= \langle Q,\ln f\rangle - \displaystyle\int \nabla_x\cdot\left(fv\right)\ln f\mathrm dv &=\langle Q,\ln f\rangle - \nabla_x \cdot \left[\int \left(f\ln f -f\right)v \mathrm dv\right]
\end{aligned}$\\

$\nabla H(f) = \displaystyle\int \nabla_x\left(f\ln f - f\right)\mathrm dv = \int \nabla_x f \ln f \mathrm dv$\\

$\partial_t H(f) + \nabla_x\displaystyle\int \left(f\ln f - f\right)v\mathrm dv = \langle Q(f,f),\ln f\rangle$\\
si on intègre en $x$ et qu'à l'infini $(f\ln f - f)v$ tend suffisamment rapidement vers 0, la formule de Stockes donne :

\[
\displaystyle\int \partial_x H(f) = \int \langle Q(f,f),\ln f\rangle \mathrm dx = \frac d{dt} \int H(f) \mathrm dx
\]

$\langle Q(f,f),\ln f\rangle = \frac14 \displaystyle\int\int\left(f'f_\star'+ff_\star\right)\left(\ln f+\ln f_\star - \ln f' - \ln f_\star'\right)B(|\!|v-v_\star|\!|)\mathrm dv \mathrm dv_\star \mathrm d\sigma$\\

\end{aligned}$\\

$\frac d{dt} \displaystyle\int H(f) \mathrm dx \leq 0$ (second principe de la thermodynamique)\\

\textbf{Equation adimentionnée pour le scaling hydrodynamique}

\[

				On part de $f(t,x,y) = f^0(X(0,t,x,y),Y(0,t,x,y))$ et $f(t,y,x) = f^0(X(0,t,y,x),Y(0,t,y,x))$.\\
				\underline{On ait ainsi ramener à la question :} A-t-on : $(Y(0,t,x,y),X(0,t,x,y)) = (X(0,t,y,x),Y(0,t,y,x))$ ?\\
				OUI
				\begin{dem}
					\begin{minipage}[t]{18.5cm}
						On montre : $(Y(s,t,x,y),X(s,t,x,y)) = (X(s,t,y,x),Y(s,t,y,x))$
					\end{minipage}\\
					\begin{minipage}[t]{18.5cm}
						\underline{Quand $s=t$ :} $\begin{aligned}[t]
							(Y(s=t,t,x,y),X(s=t,t,x,y)) = (y,x)\\
							(X(s=t,t,y,x),Y(s=t,t,y,x)) = (y,x)
						\end{aligned}$\\
						\underline{Evolution avec $s$ :} $\dfrac{d}{ds}\left[\begin{matrix}
							Y(s,t,x,y)\\
							X(s,t,x,y)
						\end{matrix}\right] = \left[\begin{matrix}
							X(z,t,x,y)\\
							Y(s,t,x,y)
						\end{matrix}\right] = \left[\begin{matrix}
							0 & 1 \\
							1 & 0
						\end{matrix}\right]\left[\begin{matrix}
							Y(s,t,x,y)\\
							X(s,t,x,y)
						\end{matrix}\right]$\\
						et $\dfrac{d}{ds}\left[\begin{matrix}
							X(s,t,y,x) \\
							Y(s,t,y,x)
						\end{matrix}\right] = \left[\begin{matrix}
						0 & 1 \\
						1 & 0
						\end{matrix}\right]\left[\begin{matrix}
						X(s,t,y,x) \\
						Y(s,t,y,x)
						\end{matrix}\right]$\\
						C'est-à-dire : \begin{minipage}[t]{15cm}
							$\left[\begin{matrix}
							Y(s,t,x,y) \\
							X(s,t,x,y)
						\end{matrix}\right]$ et $\left[\begin{matrix}
						X(s,t,y,x) \\
						Y(s,t,y,x)
						\end{matrix}\right]$\\
						satisfont la même EDO $Z' = \left[\begin{matrix}
							0 & 1 \\
							1 & 0
						\end{matrix}\right]Z$, avec la même donnée initiale en $s=t$ ces deux quantités sont égales $\forall s$.
						\end{minipage}
						
						On a ainsi montré : $(Y(0,t,x,y),X(0,t,x,y)) = (X(0,t,y,x),Y(0,t,y,x))$.\\
						Et par suite : $f(t,x,y) = f(t,y,x)$
					\end{minipage}
				\end{dem}
				
				\textbf{Remarque :} $f(t,x,y)$ et $f(t,y,x)$ satisfont la même EDP avec la même donnée initiale (car $f^0(x,y) = f^0(y,x)$) sous réserve de l'unicité de la solution à lEDP satisfaite par $f$ on peut en déduire $f(t,x,y) = f(t,y,x)$
				
				\textbf{\underline{Question 3 :}} A-t-on $f^0(x,y) = g^0(x)g^0(y) \implies f(t,x,y) = g(t,x)g(t,y)$ ?\\
				\underline{Réponse :} $f(t,x,y) = f^0(X(0,t,x,y),Y(0,t,x,y)) = g^0(x\cosh(t)-y\sinh(t))g^0(y\cosh(t)-x\sinh(t))$.\\
				on ne voit pas comment (pour une fonction $g^0$ générale) ceci pourrait se mettre sous la forme d'un produit $g(t,x)g(t,y)$.
			\end{minipage}
		\end{sol}
		
		
		
		
		
		
		
		\begin{exo}$ $
			
			\begin{minipage}[t]{18cm}
				$\begin{cases}
				x'(t) = G(x(t)-y(t))+G(x(t)-z(t))\\
				y'(t) = G(y(t)-x(t)) + G(y(t)-z(t))\\
				z'(t) = G(z(t)-x(t)) + G(z(t)-y(t))
			\end{cases}$, où $G : \R \to \R$ est donnée.
			\end{minipage}
			
			$f^0$ est donnée et on pose : $f(t,X(t,0,x,y),Y(t,0,x,y)) = f^0(x,y)$.
			
			\begin{enumerate}
				\item Quelle est l'EDP satisfaite par $f$ ?
				\item  A-t-on : $f(t,x,y) = f(t,y,x)$ ?
				\item A-t-on : $f(t,x,y) = f^1(t,x)f^1(t,y)$ ?
				\item Quelle sont les équations sur les marginales : $f^1, f^2$ et $f^3$ ?
			\end{enumerate}
		\end{exo}
		\begin{sol}$ $\\
			\begin{minipage}[t]{18.5cm}
				\textbf{\underline{Question 1 :}} quelle EDP satisfait $f$ ?\\
			On a (méthode habituelle) : 
			\[\dfrac{\partial f}{\partial t}(t,x,y,z)+\left[G(x-y)+G(x-z)\right]\dfrac{\partial f}{\partial x} + \left[G(y-x)+G(y-z)\right]\dfrac{\partial f}{\partial y} + \left[G(z-x)+G(z-y)\right]\dfrac{\partial f}{\partial z} = 0\]
			On a : $f^0(x,y,z) = f^0(y,x,z) \implies$ même propriété pour $f(t,x,y,z)$.\\
			En effet : $f(t,x,y,z) = f^0(X(0,t,x,y,z),Y(\cdots),Z(\cdots))$.\\
			Et on a : \begin{minipage}[t]{17cm}
				$(X(0,t,x,y,z),Y(0,t,x,y,z),Z(0,t,x,y,z)) = (Y(0,t,x,y,z),X(0,t,x,y,z),Z(0,t,x,y,z))$ par unicité dans Cauchy-Lipschitz.
			\end{minipage}\\
			Enfin on a : 
			\[\dfrac{\partial f}{\partial t}(t,x,y,z)+\left[G(x-y)+G(x-z)\right]\dfrac{\partial f}{\partial x} + \left[G(y-x)+G(y-z)\right]\dfrac{\partial f}{\partial y} + \left[G(z-x)+G(z-y)\right]\dfrac{\partial f}{\partial z} = 0\]
			$\left(\text{On pose } f^1(t,x) = \displaystyle\int f(t,x,y,z) dydz,\cdots,etc.\right)$\\
			
			Ainsi :
			\[\begin{aligned}
				\dfrac{\partial f^1}{\partial t}(t,x) 
			&+ \displaystyle\underbrace{\int\left[G(x-y)+G(x-z)\right]\dfrac{\partial f}{\partial x}(t,x,y,z)}_{\rouge{=2\int G(x-y)\frac{\partial f^2}{\partial t}(t,x,y) dy}} + \underbrace{\int\left[G(y-x)+G(y-z)\right]\dfrac{\partial f}{\partial x}(t,x,y,z)dydz}_{\rouge{\underset{= -\int G'(y-x)f^2(t,x,y)dy + \int G'(y-z)f(t,x,y,z)}{=-\int\left[G'(y-x)+G'(y-z)\right]f(t,x,y,z)}}}\\
			&+\cdots
			\end{aligned}\]
			\end{minipage}
		\end{sol}
