	\textbf{\underline{Maillage :}}
	\begin{enumerate}
		\item $n$ : nombre de nœuds
		\item $x_1$ $y_1$ : coordonnées du nœud 1
		\item \begin{minipage}[t]{10cm}
			$x_2$ $y_2$ : coordonnées du nœud 2
		
			$\displaystyle\vdots$
			
			 $x_n$ $y_n$ : coordonnées du nœud $n$
		\end{minipage}
		\item \begin{minipage}[t]{10cm}
			$m$ : nombre d'éléments\\
			$t$ : type des éléments\\
			$p$ : nombre de nœuds ($p.n$  éléments)
		\end{minipage}
		\item \begin{minipage}[t]{15cm}
			$s_1^{(1)}, s_2^{(1)},\ldots,s_p^{(1)}$ : liste des numéraux globaux des nœuds de l'élément 1.
		
		$\displaystyle\vdots$
		
		$s_1^{(m)}, s_2^{(m)},\ldots,s_p^{(m)}$ : liste des numéraux globaux des nœuds de l'élément $m$.
		\end{minipage}
	\end{enumerate}
