	\section{Introduction aux éléments finis}
	
	\subsection{Résolution informelles}
	
	Résoudre \begin{minipage}[t]{20cm}
		des problèmes aux limites : EDP + conditions aux bords,\\
		ou des problèmes de recherche de valeur propres d'opérateur différentiels.
	\end{minipage}\\
	
	$\star$ EDP + conditions aux bords : $\Omega \subset \R^d$ , ouvert, avec $d = 2$ ou $3$.
	
	\[\begin{cases}
		Du = f\\
		\text{conditions sur } \Gamma_D \text{ et } \Gamma_N\\
		\text{ex : } u = u_D \text{ sur } \Gamma_D
	\end{cases}\] avec $\begin{cases}
		f \text{ convexe}\\
		u_D \text{ convexe}
	\end{cases}$ et $D$ opérateur différentielle, exemple : $D = \Delta$.\\
	
	$\star$\begin{minipage}[t]{20cm}
		Valeur propres d'opérateur différentielle\\
		Chercher $(\lambda,u)$ tel que : \[\begin{cases}
			Du = \lambda u \text{ dans } \Omega\\
			\text{ avec conditions aux bord pour } u
		\end{cases}\]
	\end{minipage}\\
	
	Résolution par éléments finis :
	$\longrightarrow$ Écrire une forme faible du problème ou formulation variationnelle.
	\begin{ex}
		$\begin{cases}
			Du = f & \star \forall v : \displaystyle\int_\Omega Du \cdot v = \int_\Omega f \cdot v\\
			u = u_D \text{ sur } \partial\Omega & \star \text{ i.p.p } \displaystyle\int_\Omega D_1 u D_2 v + \underbrace{\int_{\partial\Omega} ...}_{\rouge{\text{prise en compte des cond de bord}}} = \int_\Omega f v \\
			& \text{ avec } D_1 \text{ et } D_2 \text{ opérateurs différentiels d\'ordre 1}
		\end{cases}$
	\end{ex}
	
	On arrive à un problème de la forme :
	\enc{red}{Chercher $u \in V$ tel que $\forall v \in V$ :
		\begin{center}
			$a(u,v) = l(v)$
		\end{center}
	}
	avec \begin{minipage}[t]{15cm}
		$(a(\cdot,\cdot)$ forme bilinéaire.\\
		$l(\cdot)$ forme linéaire.
	\end{minipage}
	
	\vspace{.5cm}
	
	$\longrightarrow$ \begin{minipage}[t]{20cm}
		Discrétisation
		\begin{enumerate}
			\item[$\star$] géométrique $\Omega \to \Omega_h$ maillage.
			\item[$\star$] de l'inconnue. $V$ espace fonctionnel de dimension infini et $V_h$ de dimension fini.
		\end{enumerate}
	\end{minipage}
	
	$V \to V_h$ dimension finie $V_h = Vect \{\phi_I, I = 1,\ldots,N\}$\\
	$u_h$ dans $V_h$ s'exprime comme combinaison des $\phi_I$, $I = 1,\ldots,N$.
	\[
	u_h = \displaystyle\sum_{j+1}^N u_j \phi_J \text{ avec } U = (u_1, \ldots, u_N) \in \R^N
	\]
	
	On écrit l'expression $(FV)$ pour $v = \phi_I$ :\\
	$\begin{cases}
		\forall I = 1,\ldots, N\\
		a(u_h,\phi_I) = l(\phi_I)
	\end{cases}$ c'est-à-dire $\begin{cases}
		\forall I = 1,\ldots,N\\
		\displaystyle\sum_{J=1}^N \underbrace{a(\phi_J,\phi_I)}_{A_{I,J}}u_J = \underbrace{l(\phi_I)}_{B_I}
	\end{cases}$\\
	On obtient donc \fbox{$AU = B$}.
	
	\subsection{Établissement de la formulation variationnelle}
	\subsubsection{Les espaces fonctionnels}
	Notation : $\m C^{0,\alpha}(\Omega) = $ l'ens. des fonctions $f \in \m C^0$ sur $\Omega$ tel que :
	\hfill $\forall x,y \in \Omega, $ $\|f(x)-f(y)\| \leq \| x-y\|^\alpha$
	
	\begin{defi}
		Soit $\Omega \subset \R^d$, un ouvert, $\Omega$ est dit $\m C ^k$ (resp. $\m C^{k,1}$).\\
		Si $\forall m_0 \in \partial \Omega$, $\exists B(m_0,R)$ avec $R > 0$ et un système d'axes $\{0_0,x_1^0,\ldots,x_d^0\}$ et $q_0 \in \m C^k(\R^{d-1})$ (resp. $\m C^{k,1}$) tq :\\
		$\partial\Omega\cap B(m_0,R) = \{x\in B(m_0,R); x_d^0 = a_0(x_1^0,\ldots,x_{d-1}^0)\}$\\
		$\Omega\cap B(m_0,R) = \{x\in B(m_0,R); x_d^0 > a_0(x_1^0,\ldots,x_{d-1}^0)\}$
	\end{defi}
	
	On va considérer des espaces de Sobolev.
	\begin{ex}
		$H^1(\Omega) = \left\{f \in \text{L}^2(\Omega); \dfrac{\partial}{\partial x_i} f \in \text{L}^2(\Omega), i=1,\ldots,d\right\}$
	\end{ex}

	\subsubsection{Espace de Sobolev}
	
	\begin{minipage}{6cm}
		\enc{blue}{Notion de distribution}
	\end{minipage}
	
	\textbf{\underline{Notation :}} \begin{minipage}{15cm}
		$\partial^\alpha u = \dfrac{\partial^{|\alpha|}u}{\partial_1^{\alpha_1}\ldots\partial_d^{\alpha_d}}$, avec $\alpha = (\alpha_1,\ldots,\alpha_d) \in \N^d$ et $|\alpha| = \alpha_1 + \ldots + \alpha_d$
	\end{minipage}
	
	\begin{defi}
		$\Omega$ ouvert de $\R^d$.
		\begin{enumerate}
			\item[$\star$] $K$ compact dans $\Omega$, si $K \subset \Omega$ et $K$fermé borné.
			\item[$\star$] $\begin{aligned}[t]
				\m D(\Omega)
				&= \text{ensemble dit des "fonctions tests".}\\
				&= \text{ensemble des fonctions à support compact dans } \Omega \text{ et }
 \m C^\infty \text{ sur } \Omega
 			\end{aligned}$
			\item[$\star$] $u$ est dit une distribution sur $\Omega$ :\\
	
			\hspace*{-1cm}\begin{minipage}{15cm}
				Si $u$ est une forme linéaire sur $\m D (\Omega)$ et si $u$ vérifie une propriété de continuité :
			\begin{center}
				$
			\forall K \text{ compact de } \Omega, \exists k \in \N \text{ et } \exists C_K \text{ tq : }
			$
			\hspace{3cm}$
			\forall\phi\in \m D(\Omega) \text{ tq } \text{supp}(\phi) \subset K, |<u,\phi>| \leq C_K \max_{|\alpha| \leq k}\{\|\partial^\alpha\|\}
		 	$
			\end{center}
			\end{minipage}\hspace{-.5cm}\begin{minipage}{6.7cm}
			\enc{red}{$u : \phi \in \m D(\Omega) \mapsto \underbrace{<u,\phi>}_{\text{produit dualité}} \in \R$}
			\end{minipage}
			
			\item[$\star$] On note $\m D'(\Omega)$ l'ensemble des distributions sur $\Omega$.
		\end{enumerate}
	\end{defi}
	
	\begin{ex} 
		\begin{minipage}[t]{20cm}
			
			\vspace{-.2cm}
			
			\begin{enumerate}
			\item $x_0 \in \Omega$.\\
			$\delta_{x_0} : \phi \mapsto \phi(x_0)$, \textbf{linéaire en $\phi$}
			
			$\delta_{x_0} \in \m D'(\Omega)$, \begin{minipage}[t]{15cm}
				$|<\delta_{x_0}, \phi>| =  |\phi(x_0)| \leq \|\phi\|_{\text{L}^infty(\Omega)}$\\
				\textbf{Propriété de continuité} avec $C_K = 1$ et $k = 0$, $\forall K$.
				\end{minipage}
				
				\vspace{.5cm}
				
			\item Soit $f \in \text{L}^2(\Omega)$.\\
			Soit $u_f : \phi \mapsto \displaystyle\int_\Omega f\phi$, \begin{minipage}[t]{12cm}
				\textbf{linéaire en $\phi$}\\
				et $\begin{aligned}[t]
					\abs{<u_f, \phi>} 
					&= \abs{\int_\Omega f \phi} \\
					&\leq \|f\|_{\text{L}^2(\Omega)}\|\phi\|_\text{L}^2(\Omega) \\
					&\leq C\|f\|_{\text{L}^2(\Omega)}\|\phi\|_{\text{L}^2(\Omega)}
				\end{aligned}$
			\end{minipage}
			
			Donc $u_f$ est une distribution.
			
		\end{enumerate}
		\end{minipage}
	\end{ex}

	\begin{minipage}{10cm}
		\enc{blue}{Convergence et dérivation au sens des distributions}
	\end{minipage}
	
	\vspace{.5cm}
	
	$\star$ \begin{minipage}[t]{20cm}
				\underline{Convergence}\\
				Soient $(u_n)_{n \i \N}$ une suite de distributions et $u$ une distribution.\\
				On dit que $(u_n)_n$ tend vers $u$ au sens des distributions si :
				\[
				<u_n,\phi> \to <u,\phi>, \forall \phi \in \m D(\Omega)
				\]
			\end{minipage}
	\par $\star$ \begin{minipage}[t]{20cm}
					Dérivation : soit $u \in \m D'(\Omega)$, la dérivée de $u$ au sens des distribution, $\dfrac{\partial u}{\partial x_i}, i = 1,\ldots,d$ est définie par :
					\[
					<\dfrac{\partial u}{\partial x_i},\phi> = - <u,\dfrac{\partial\phi}{\partial x_i}>, \forall \phi \in \m D (\Omega)
					\]
				\end{minipage}
	
	\par $\star$ \begin{minipage}[t]{20cm}		
					$\alpha$ multi-indice.\\
					$<\partial^\alpha u,\phi> = (-1)^{|\alpha|}<u,\partial^\alpha\phi>, \forall\phi\in \m D(\Omega)$
				\end{minipage}
	
	\begin{ex}
		\begin{minipage}[t]{15cm}
			Si $f \in \text{L}^2(\Omega)$, et $u_f$ la distribution associée : $u_f : \phi \mapsto \int_\Omega f \phi$.
		\end{minipage}
		
			\[\begin{aligned}
			<\dfrac{\partial}{\partial x_i} u_f,\phi> 
			&= -<u_f, \frac{\partial\phi}{\partial x_i}>\\ 
			&= - \displaystyle\int_\Omega f\dfrac{\partial\phi}{\partial x_i} \\
			&= \int_\Omega\dfrac{\partial f}{\partial x_i} \phi + \underbrace{\text{terme de bord}}_{= 0 \text{ car } \phi \in \m D(\Omega)} \\
			&= <u_{\frac{\partial f}{\partial x_i}},\phi>
		\end{aligned}\]
 	\end{ex}
 	
 	\textbf{\underline{Remarque :}} \begin{minipage}[t]{15cm}
								 			Par abus de langage et de notation, on dit que $f \in \text{L}^2(\Omega)$ est dérivable au sens des distributions et on note $\dfrac{\partial f}{\partial x_i}$ la dérivée au sens des distributions de la distribution associée à $f$.
								 	\end{minipage}
