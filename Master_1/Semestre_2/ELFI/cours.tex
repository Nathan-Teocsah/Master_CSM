	\section{Introduction aux éléments finis}
	
	\subsection{Résolution informelles}
	
	Résoudre \begin{minipage}[t]{20cm}
		des problèmes aux limites : EDP + conditions aux bords,\\
		ou des problèmes de recherche de valeur propres d'opérateur différentiels.
	\end{minipage}\\
	
	$\star$ EDP + conditions aux bords : $\Omega \subset \R^d$ , ouvert, avec $d = 2$ ou $3$.
	
	\[\begin{cases}
		Du = f\\
		\text{conditions sur } \Gamma_D \text{ et } \Gamma_N\\
		\text{ex : } u = u_D \text{ sur } \Gamma_D
	\end{cases}\] avec $\begin{cases}
		f \text{ convexe}\\
		u_D \text{ convexe}
	\end{cases}$ et $D$ opérateur différentielle, exemple : $D = \Delta$.\\
	
	$\star$\begin{minipage}[t]{20cm}
		Valeur propres d'opérateur différentielle\\
		Chercher $(\lambda,u)$ tel que : \[\begin{cases}
			Du = \lambda u \text{ dans } \Omega\\
			\text{ avec conditions aux bord pour } u
		\end{cases}\]
	\end{minipage}\\
	
	Résolution par éléments finis :
	$\longrightarrow$ Écrire une forme faible du problème ou formulation variationnelle.
	\begin{ex}
		$\begin{cases}
			Du = f & \star \forall v : \displaystyle\int_\Omega Du \cdot v = \int_\Omega f \cdot v\\
			u = u_D \text{ sur } \partial\Omega & \star \text{ i.p.p } \displaystyle\int_\Omega D_1 u D_2 v + \underbrace{\int_{\partial\Omega} ...}_{\rouge{\text{prise en compte des cond de bord}}} = \int_\Omega f v \\
			& \text{ avec } D_1 \text{ et } D_2 \text{ opérateurs différentiels d\'ordre 1}
		\end{cases}$
	\end{ex}
	
	On arrive à un problème de la forme :\\
	\enc{red}{Chercher $u \in V$ tel que $\forall v \in V$ :
		\begin{center}
			$a(u,v) = l(v)$
		\end{center}
	avec \begin{minipage}[t]{15cm}
		$a(\cdot,\cdot)$ forme bilinéaire.\\
		$l(\cdot)$ forme linéaire.
	\end{minipage}
	}
	
	
	\newpage
	
	$\longrightarrow$ \begin{minipage}[t]{20cm}
		\textbf{\underline{Discrétisation}}
		\begin{enumerate}
			\item[$\star$] géométrique $\Omega \to \Omega_h$ maillage.
			\item[$\star$] de l'inconnue. $V$ espace fonctionnel de dimension infini et $V_h$ de dimension fini.
		\end{enumerate}
	\end{minipage}
	
	\vspace{.5cm}
	
	\underline{\textit{On transforme le problème :}} $V \to V_h = \text{Vect} \{\phi_I, I = 1,\ldots,N\}$ en dimension finie.
	
	$u_h$ dans $V_h$ s'exprime comme combinaison des $\phi_I$, $I = 1,\ldots,N$.
	\[
	u_h = \displaystyle\sum_{j+1}^N u_j \phi_J \text{ avec } U = (u_1, \ldots, u_N) \in \R^N
	\]
	
	On écrit l'expression $(FV)$ (formulation variationnelle) pour $v = \phi_I$ :
	
	$\begin{cases}
		\forall I = 1,\ldots, N\\
		a(u_h,\phi_I) = l(\phi_I)
	\end{cases}$ c'est-à-dire $\begin{cases}
		\forall I = 1,\ldots,N\\
		\displaystyle\sum_{J=1}^N \underbrace{a(\phi_J,\phi_I)}_{A_{I,J}}u_J = \underbrace{l(\phi_I)}_{B_I}
	\end{cases}$
	
	On obtient donc \fbox{$AU = B$}.

	\subsubsection{Les espaces fonctionnels}
	\underline{\textbf{Notation :}} $\m C^{0,\alpha}(\Omega) = $ l'ens. des fonctions $f \in \m C^0$ sur $\Omega$ tel que :
	\hfill $\forall x,y \in \Omega, $ $\|f(x)-f(y)\| \leq \| x-y\|^\alpha$
	
	\begin{defi}
		Soit $\Omega \subset \R^d$, un ouvert, $\Omega$ est dit $\m C ^k$ (resp. $\m C^{k,1}$).\\
		Si $\forall m_0 \in \partial \Omega$, $\exists B(m_0,R)$ avec $R > 0$ et un système d'axes $\{0_0,x_1^0,\ldots,x_d^0\}$ et $q_0 \in \m C^k(\R^{d-1})$ (resp. $\m C^{k,1}$) tq :\\
		$\partial\Omega\cap B(m_0,R) = \{x\in B(m_0,R); x_d^0 = a_0(x_1^0,\ldots,x_{d-1}^0)\}$\\
		$\Omega\cap B(m_0,R) = \{x\in B(m_0,R); x_d^0 > a_0(x_1^0,\ldots,x_{d-1}^0)\}$
	\end{defi}
	
	On va considérer des espaces de Sobolev.
	\begin{ex}
		$H^1(\Omega) = \left\{f \in \text{L}^2(\Omega); \dfrac{\partial}{\partial x_i} f \in \text{L}^2(\Omega), i=1,\ldots,d\right\}$
	\end{ex}

	\subsubsection{Notion de distribution}
	
	\textbf{\underline{Notation :}} \begin{minipage}{15cm}
		$\partial^\alpha u = \dfrac{\partial^{|\alpha|}u}{\partial_1^{\alpha_1}\ldots\partial_d^{\alpha_d}}$, avec $\alpha = (\alpha_1,\ldots,\alpha_d) \in \N^d$ et $|\alpha| = \alpha_1 + \ldots + \alpha_d$
	\end{minipage}
	
	\begin{defi}
		$\Omega$ ouvert de $\R^d$.
		\begin{enumerate}
			\item[$\star$] $K$ compact dans $\Omega$, si $K \subset \Omega$ et $K$fermé borné.
			\item[$\star$] $\begin{aligned}[t]
				\m D(\Omega)
				&= \text{ensemble dit des "fonctions tests".}\\
				&= \text{ensemble des fonctions à support compact dans } \Omega \text{ et }
 \m C^\infty \text{ sur } \Omega
 			\end{aligned}$
			\item[$\star$] $u$ est dit une distribution sur $\Omega$ :\\
			Si $u$ est une forme linéaire sur $\m D (\Omega)$ et si $u$ vérifie une propriété de continuité :
			
			\begin{minipage}{7.5cm}
				
			$\forall K \text{ compact de } \Omega, \exists k \in \N \text{ et } \exists C_K \text{ tq : }$
			\begin{center}
			$\forall\phi\in \m D(\Omega) \text{ tq } \text{supp}(\phi) \subset K,$\\ $\abs{\left<u,\phi\right>} \leq C_K \max_{|\alpha| \leq k}\{\|\partial^\alpha\phi\|\}$
			\end{center}
			
			\end{minipage}\begin{minipage}{6.7cm}
			$ $ \hspace*{2cm}\enc{red}{$u : \phi \in \m D(\Omega) \mapsto \underbrace{\left<u,\phi\right>}_{\text{produit dualité}} \in \R$}
			\end{minipage}
			
			\item[$\star$] On note $\m D'(\Omega)$ l'ensemble des distributions sur $\Omega$.
		\end{enumerate}
	\end{defi}
	
	\begin{ex} 
		\begin{minipage}[t]{20cm}
			
			\vspace{-.2cm}
			
			\begin{enumerate}
			\item $x_0 \in \Omega$.\\
			$\delta_{x_0} : \phi \mapsto \phi(x_0)$, \textbf{linéaire en $\phi$}
			
			$\delta_{x_0} \in \m D'(\Omega)$, \begin{minipage}[t]{15cm}
				$\abs{\left<\delta_{x_0}, \phi\right>} =  \abs{\phi(x_0)} \leq \|\phi\|_{\text{L}^infty(\Omega)}$\\
				\textbf{Propriété de continuité} avec $C_K = 1$ et $k = 0$, $\forall K$.
				\end{minipage}
				
				\vspace{.5cm}
				
			\item Soit $f \in \text{L}^2(\Omega)$.\\
			Soit $u_f : \phi \mapsto \displaystyle\int_\Omega f\phi$, \begin{minipage}[t]{12cm}
				\textbf{linéaire en $\phi$}\\
				et $\begin{aligned}[t]
					\abs{\left<u_f, \phi\right>} 
					&= \abs{\int_\Omega f \phi} \\
					&\leq \|f\|_{\text{L}^2(\Omega)}\|\phi\|_\text{L}^2(\Omega) \\
					&\leq C\|f\|_{\text{L}^2(\Omega)}\|\phi\|_{\text{L}^2(\Omega)}
				\end{aligned}$
			\end{minipage}
			
			Donc $u_f$ est une distribution.
			
		\end{enumerate}
		\end{minipage}
	\end{ex}

	\subsubsection{Convergence et dérivation au sens des distributions}
	
	$\star$ \begin{minipage}[t]{20cm}
				\underline{Convergence}\\
				Soient $(u_n)_{n \i \N}$ une suite de distributions et $u$ une distribution.\\
				On dit que $(u_n)_n$ tend vers $u$ au sens des distributions si :
				\[
				\left<u_n,\phi\right> \to \left<u,\phi\right>, \forall \phi \in \m D(\Omega)
				\]
			\end{minipage}
	\par $\star$ \begin{minipage}[t]{20cm}
					\underline{Dérivation :} soit $u \in \m D'(\Omega)$, la dérivée de $u$ au sens des distribution, $\dfrac{\partial u}{\partial x_i}, i = 1,\ldots,d$ est définie par :
					\[
					\left<\dfrac{\partial u}{\partial x_i},\phi\right> = - \left<u,\dfrac{\partial\phi}{\partial x_i}\right>, \forall \phi \in \m D (\Omega)
					\]
				\end{minipage}
	
	\par $\star$ \begin{minipage}[t]{20cm}		
					\underline{$\alpha$ multi-indice}.\\
					$\left<\partial^\alpha u,\phi\right> = (-1)^{|\alpha|}\left<u,\partial^\alpha\phi\right>, \forall\phi\in \m D(\Omega)$
				\end{minipage}
	
	\begin{ex}
		\begin{minipage}[t]{15cm}
			Si $f \in \text{L}^2(\Omega)$, et $u_f$ la distribution associée : $u_f : \phi \mapsto \int_\Omega f \phi$.
		\end{minipage}
		
			\[\begin{aligned}
			\left<\dfrac{\partial}{\partial x_i} u_f,\phi\right> 
			&= -\left<u_f, \frac{\partial\phi}{\partial x_i}\right>\\ 
			&= - \displaystyle\int_\Omega f\dfrac{\partial\phi}{\partial x_i} \\
			&= \int_\Omega\dfrac{\partial f}{\partial x_i} \phi + \underbrace{\text{terme de bord}}_{= 0 \text{ car } \phi \in \m D(\Omega)} \\
			&= \left<u_{\frac{\partial f}{\partial x_i}},\phi\right>
		\end{aligned}\]
 	\end{ex}
 	
 	\textbf{\underline{Remarque :}} \begin{minipage}[t]{15cm}
								 			Par abus de langage et de notation, on dit que $f \in \text{L}^2(\Omega)$ est dérivable au sens des distributions et on note $\dfrac{\partial f}{\partial x_i}$ la dérivée au sens des distributions de la distribution associée à $f$.
								 	\end{minipage}
								 	
	\begin{ex}
		\begin{minipage}[t]{15cm}
			\begin{minipage}[t]{10cm}
			$f$ dans $\text{L}^2(]0,2[)$ définie par $f(x) = \begin{cases}
															0 \text{ sur } ]0,1]\\
															1 \text{ sur } ]1, 2]
														\end{cases}$.
		\end{minipage} \begin{minipage}{2cm}
														\includegraphics[scale = .4]{photo1.png}
														\end{minipage}
		\end{minipage}
		
		$\begin{aligned}
			\left<\dfrac{\partial f}{\partial x_1},\phi\right> 
			&= -\left<f,\dfrac{\partial\phi}{\partial x_1}\right> 
			&= - \displaystyle\int f\dfrac{\partial\phi}{\partial x_1} 
			&= -\int_1^2\dfrac{\partial\phi}{\partial x_1} \\
			& & & = - \left[\phi\right]_1^2\\
			& & & = \phi(1)\\
			& & & = \left<\delta_1,\phi\right> \not \in \text{L}^2(\Omega)
		\end{aligned}$.
	\end{ex}
	
	\subsubsection{Espaces de Sobolev}
	
	\begin{defi}
		$H^1(\Omega) = \left\{f \in  \text{L}^2(\Omega); \dfrac{\partial f}{\partial x_i} \in \text{L}^2(\Omega), i=1,\ldots,d \right\}$, 
		
		\vspace{.2cm}
		
		équipé du produit scalaire (sur $\m C^1$) :
		\[
		\left<u,v\right>_{H^1} = \displaystyle\int_\Omega uv + \int_\Omega \nabla u\cdot\nabla v
		\]
	\end{defi}
	
	\textbf{\underline{Remarque :}} $\displaystyle\int_\Omega \nabla u\cdot\nabla v = \sum_{i=1}^N \int_\Omega \dfrac{\partial u}{\partial x_i}\dfrac{\partial v}{\partial x_i}$
	
	\begin{prop}
		$(H^1(\Omega), \left<\cdot,\cdot\right>_{H^1(\Omega)})$ est un espace de Hilbert.
	\end{prop}
	
	\begin{defi}
		Soit $m \in \N$.\\
		\begin{enumerate}
			\item[$\star$] $\underline{H^m(\Omega)} = \left\{f \in L^2(\Omega); \partial^\alpha f\in L^2(\Omega), \forall\alpha\in \N^d \text{ tq } |\alpha| \leq m\right\}$, équipé du produit scalaire : 
			\[
			\left< u,v \right>_{H^m(\Omega)} = \displaystyle\sum_{|\alpha|\leq m} \int_\Omega \partial^\alpha m . \partial^\alpha v
			\]
			et de la norme associée : $\|u\|_{H^m(\Omega)} = \left(\left< u,u \right>\right)^{1/2}$.
			
			\item[$\star$] \underline{$H_0^1(\Omega)$} est le complété de $\m D(\Omega)$ dans $(H^1(\Omega),\|\cdot\|_{H^1(\Omega)})$.\\
			On note $\underbrace{H^{-1}(\Omega)}_{\text{sous ensemble de } \m D'(\Omega)} = \left(H_0^1(\Omega)\right)'$\\
			c'est l'ensemble des formes linéaire définie sur $H^1_0(\Omega)$.
		\end{enumerate}
	\end{defi}
	
	\begin{prop}
		$(H^m(\Omega),\left<\cdot,\cdot\right>_{H^m(\Omega)})$ est aussi un espace de Hilbert.
	\end{prop}
	
	\begin{defi}[Opérateur trace]
		Soit $\Omega$ un ouvert de $\R^d$.\\
		On note $\gamma_0$ l'opérateur trace sur le bord :
		\[
		\begin{aligned}
			\gamma_0 \text{ : }& H^1(\Omega) & \rightarrow & & L^2(\partial \Omega)\\
			 & f & \mapsto & & \gamma_0 f = f_{\partial \Omega}
		\end{aligned}
		\]
	\end{defi}
	
	\begin{thm} $ $
		\begin{enumerate}
			\item[$\star$] \begin{minipage}[t]{15cm}
				$H^1_0(\Omega) = \left\{u\in H^1(\Omega); \overset{\sim}{u}\in H^1(\R^d)\right\}$, avec $\overset{\sim}{u}$ définie par $\overset{\sim}{u} = 0$ sur $\R^d\backslash\Omega$
			\end{minipage}
			
			\vspace{.5cm}
			
			\item[$\star$] $H^1_0(\Omega) = \ker \gamma_0 = \left\{u\in H^1(\Omega); \gamma_0 u = 0 \text{ sur } \partial\Omega\right\}$
		\end{enumerate}
	\end{thm}
	
	\textbf{\underline{Notation :}} \begin{minipage}[t]{15cm}
		$H^{3/2}(\partial\Omega) = \left\{\underline{u} \in L^2(\partial\Omega); \exists u \in H^2(\Omega)  \text{ avec } \underline{u} = \gamma_0 u\right\}$
		
		\vspace{.5cm}
		
		$H^{1/2}(\partial\Omega) = \left\{\underline{u} \in L^1(\partial\Omega); \exists u \in H^2(\Omega)  \text{ avec } \underline{u} = \gamma_0 u\right\}$
	\end{minipage}
	
	
	\subsubsection{Formule de Green}
	
	\begin{thm}[Formule de Green]
		\begin{minipage}[t]{7cm}
			Soit $\Omega$ est un ouvert de $\R^d$.\\
		$\forall u,v \in H^1(\Omega)$, 
		\end{minipage}\begin{minipage}{6cm}
		\includegraphics[scale=.3]{photo2.png}
		\end{minipage}
		\[
		\displaystyle\int_\Omega \dfrac{\partial u}{\partial x_i} v = - \int_\Omega u \dfrac{\partial v}{\partial x_i} + \int_\Omega\gamma_0 u .\gamma_0 v .\nu_i 
		\]
		pour $ i = 1, \ldots, d$.\\
		
		\begin{minipage}{16cm}
			Avec $\nu_i$ $i$-ième composante de $\nu$ le vecteur unitaire normal à $\partial \Omega$ sortant par rapport à $\Omega$.
		\end{minipage}
	\end{thm}
	
	\begin{cor}
		Soit $\Omega \subset \R^d$ est $\m C^{0,1}$, $u \in H^1(\Omega)$ et $v \in H^1(\Omega)$.
		\[
		\displaystyle\int_\Omega \Delta u v = - \int_\Omega \nabla u \cdot \nabla v + \int_{\partial\Omega} \dfrac{\partial u}{\partial\nu} v
		\]
		avec $\dfrac{\partial u}{\partial \nu} = \nabla \cdot \nu$.
	\end{cor}
	
	\begin{rmq}
		 $\nabla u = \left(\dfrac{\partial u}{\partial x_1},\ldots,\dfrac{\partial u}{\partial x_d}\right)$ et $\nabla \cdot \nu = \left(\gamma_0 \dfrac{\partial u}{\partial x_1},\ldots,\gamma_0 \dfrac{\partial u}{\partial x_1}\right)$.\\
		 On omet le $\gamma_0$ à partir de maintenant, il est "sous-entendu".
	\end{rmq}
	
	\subsubsection{EDP $\rightarrow$ Formulation variationnelle, caractère bien posé}
	
	\begin{minipage}[t]{5cm}
		\begin{center}
			EDP\\
			+\\
			condition aux bords
		\end{center}
	\end{minipage}\begin{minipage}[t]{16cm}
	
	\vspace{.2cm}
	
	$\xrightarrow[\text{EDP}\times v]{} $ $\cdots$ $\xrightarrow[\text{formule de Green}]{}$ $\cdots$ $\xrightarrow[\text{Prise en compte des conitions aux bords}]{}$ (FV)
	\end{minipage}
	
	(FV) : \fbox{\begin{minipage}{5cm}
		Chercher $u \in V$, tq $\forall v \in V$,
		\[
		a(u,v) = l(v)
		\]
	\end{minipage}}

	\hspace{1cm} $a$ forme bilinéaire continue sur $V \times V$ et $l$ forme linéaire continue sur $V$.
	
	\begin{lm}[Lax-Milgram]
		\begin{minipage}[t]{15cm}
			Si, de plus, $a$ est coercive : $\exists \alpha > 0 \text{ tq } \forall u \in V, a(u,v) \geq \alpha \|u\|^2$,
		\end{minipage}
		
		$\longrightarrow$ alors il existe un solution de la formulation variationnelle.
	\end{lm}

	\section{Problème modèle}
	
	\subsection{Formulation variationnelle}
	\begin{enumerate}
		\item[$\star$] On multiplie l'EDP par une fonction et on intègre sur $\Omega$, fonction test $v \in H^1(\Omega)$.
		\[
		\displaystyle -\int_\Omega \Delta u v + \int_\Omega uv = \int_\Omega f_\Omega v
		\]
		
		\item[$\star$] Formule de Green\\
		c'est-à-dire :$\displaystyle \int_\Omega \Delta u v = - \int_\Omega \nabla u \cdot \nabla v + \int_{\partial \Omega}\dfrac{\partial u}{\partial v} v$.\\
		\begin{minipage}[t]{10cm}
			Ainsi :
		\[
		\displaystyle \int_\Omega \nabla u \cdot \nabla v + \int_\Omega uv = \int_\Omega f_\Omega v + \int_{\partial\Omega} \dfrac{\partial u}{\partial \nu}v
		\]
		\end{minipage}
		
		\item[$\star$] Prise en compte des conditions aux bords : $u$ est donnée sur $\Gamma_D$.\\
		On pose $v = 0$ sur $\Gamma_D$.\\
		Ainsi : $\displaystyle\int_{\partial\Omega} \dfrac{\partial u}{\partial\nu} v \underbrace{=}_{v = 0 \text{ sur } \Gamma_D} \int_{\Gamma_N} \dfrac{\partial u}{\partial\nu} v = \int_{\Gamma_N} \rouge{g_N} v$, \scriptsize{\rouge{condition sur $\Gamma_N$}}
	\end{enumerate}
	
	\begin{minipage}{20cm}
		La formulation variationnelle s'écrit :\\
		Chercher $u \in H^1(\Omega)$ tel que $u = u_D$ sur $\Gamma_D$ et tel que $\forall v \in H^1(\Omega), v = 0$ sur $\Gamma_D$.	
	
		\[
		\underbrace{\int_\Omega \nabla u \cdot \nabla v}_{a(u,v)} + \int_\Omega uv = \int_\Omega f_\Omega v + \underbrace{\int_{\Gamma_N} g_N \gamma_0 v}_{l(v)}
		\]
		
		$a$ est coercive : $|a(u,u)| = \|u\|_{H^1(\Omega)}^2$.
	\end{minipage}

	- $\m T_h$ est la triangulation \hspace{1cm} - $\m T_h$ est un ensemble de triangles.\\

	\begin{rmq}
		diam($K$) est le supremum de la distance entre deux points de $K$.
	\end{rmq}
	
	\begin{defi}[$\mathbb{P}_1$ et $\mathbb{Q}_1$]
		On définit :\\
		\begin{minipage}{18cm}
			$\mathbb{P}_1 =$ \begin{minipage}[t]{16cm}
				ensemble des polynôme de degré au plus 1 par rapport à \textbf{l'ensemble des} composantes.\\
				Pour les \underline{triangles}.
			\end{minipage}\\
			$\mathbb{Q}_1 =$ \begin{minipage}[t]{15cm}
				ensemble des polynôme de degré au plus 1 par rapport à \textbf{chaque} composante.\\
			Pour les \underline{quadrangle}.
			\end{minipage}
		\end{minipage}
		
		\vspace{-.5cm}
		
	\end{defi}
	
	\begin{rmq}
		\begin{minipage}[t]{15cm}
			$p \in \mathbb{P}_1$ si $p(x) = a_0 + a_1x_1 + a_2 x_2+ \cancel{a_3x_1x_2}$\\
		$q \in \mathbb{Q}_1$ si $q(x) = a_0 + a_1x_1 + a_2x_2 + a_3x_1x_2$
		\end{minipage}
	\end{rmq}

		\begin{defi}
			Pour $K \in \mathcal{T}_h$, on définie 
			\[\begin{aligned}
				a_K(u_h,v_h) =& \displaystyle\int_K \nabla u_h \cdot \nabla v_h + \int_K u_hv_h\\
				l_K(v_h) =& \int_K f_\Omega v_h + \int_{\partial K\cap \Gamma_{N,h}} g_{N,h}v_h
			\end{aligned}\]
		\end{defi}
		
		\begin{rmq}
			\begin{minipage}[t]{12cm}
				$a_h(u_h,v_h) = \displaystyle\sum_{K\in \mathcal{T}_h} a_K(u_h,v_h)$ et $l_h(v_h) = \displaystyle\sum_{K\in \mathcal{T}_h} l_K(v_h)$\\
				$a_h(w_J,w_I) = \displaystyle\sum_{K\in \mathcal{T}_h}a_K(w_J,w_I)$
			\end{minipage}
			
			\hspace{-.5cm}Si $n_J \in K$, alors $\exists j$ tel que $w_J = w_j^K$\\
			Si $n_I \in K$, alors $\exists i$ tel que $w_I = w_i^K$.\\
			\underline{Question :} Comment calculer (informatiquement) efficacement :
			\[a_h(w_J,w_I) = \displaystyle\sum_{K \in\mathcal{T}_h(n_I)\cap\mathcal{T}_h(n_J))} \sum_{i=1}^{N_K}\sum_{j=1}^{N_K} \delta_I^{g^K(i)}\delta^{g^K(j)} a_K(w_j^K,w_i^K)\]
			
			\underline{Réponse :} On ne fait pas "bêtement" une boucle sur chacun des indices (impossible à calculer en temps raisonnable).
			\begin{algorithm}[H]
				\caption{Comment calculer $a_h(w_J,w_I)$}\label{alg:cap}
				\begin{algorithmic}
					\For{$K$}
						\For{$i = 1,\cdots,N_K$}
							\State $I \gets g^K(i)$
							\For{$j = 1,\cdots,N_K$}
								\State $J \gets g^K(j)$
								\State $a_h(w_J,w_I) \gets a_K(w_j^K,w_i^K) + a_h(w_J,w_I)$
							\EndFor
						\EndFor
					\EndFor
				\end{algorithmic}
			\end{algorithm}
		\end{rmq}

	\section{Éléments finis usuels}
	
	\subsection{Définition d'un élément finis}
		Soit $K$ convexe fermé sous-ensemble de $\R^2$, avec $\overset\circ K \not= \emptyset$.\\
		Soit $P$ un espace de dimension finie de foncions de $K$ dans $\R$.\\
		Soient $N$ formes linéaires $\Phi_i$ définie sur $P$, $\Sigma = \{\Phi_i,i=1,\cdots,N\}$.
		
		\begin{defi}
			$\Sigma$ est dit $P$-unisolvante si $\Sigma$ et $l$ vérifient : \\
			\hspace*{1cm} \begin{minipage}[t]{18.5cm}
				soit $\{\alpha_i\}_{i=1,\cdots,N}$ alors il \textbf{existe un unique} $p \in P$ tel que $\Phi_i(p) = \alpha_i$ pour $i \in [\![1,N]\!]$.
			\end{minipage}
		\end{defi}
		
		\begin{prop}
			On a équivalence entre les propositions suivantes :
			\begin{enumerate}[label=$\star$]
				\item $\Sigma$ est $P$-unisolvante.
				
				\item $\begin{cases}
					\forall i = [\![1, N]\!] \Phi_i(p) = 0 \implies p = 0\\
					\dim(P) = N
				\end{cases}$
				
				\item $\begin{cases}
					\dim(P) = N \\
					\exists N \text{ fonctions } (p_i)_{i\in[\![1,N]\!]} \text{ tel que } \forall j \in [\![1,N]\!], \Phi_j(p_i) = \delta_{ij}
				\end{cases}$
			\end{enumerate}
		\end{prop}
		
		\begin{defi}
			\begin{minipage}[t]{18.5cm}
				Un triplet $(K,P,\Sigma)$ tel que $\Sigma$ est $P$-unisolvante est appelé un élément fini.
			\end{minipage}
		\end{defi}
		
		\begin{ex}$ $
				
			\begin{enumerate}
				\item \begin{minipage}[t]{18.5cm}
					En 2D, $K$ un triangle avec 3 nœuds, $n_i^K$ étant les sommets de $K$.\\
				$\Sigma$ définie par morceaux : $N = 3$, $\Phi_i : \begin{aligned}[t]
					\mathbb{P}_1(K) & \to \R \\
					p & \mapsto p(n_i^K)
				\end{aligned}$ ($P = \mathbb P_1(K)$)\\
				$\to$ Éléments fini $\mathbb P_1$ de Lagrange conforme du \textbf{Chapitre précédents}.
				\end{minipage}
				\item  \begin{minipage}[t]{18.5cm}
					En 2D, $K$ un triangle ou quadrangle droit.\\
					$\Phi_i : p \mapsto p(n_i^K)$ où $n_i^K \in K$ "arbitraire"\\
					Élément fini $\Big|_\text{de Lagrange}\hspace{-1.5cm}\text{nodal}$\\
					degré de liberté nodal.
				\end{minipage}
				
				\item \begin{minipage}[t]{18.5cm}
					$\Phi_i : p \mapsto Dp(n_i^K)v_i$ \begin{minipage}[t]{15cm}
						où $n_i^K \in K$ sont les nœuds \\
						et $v_i$ des vecteurs "arbitraires".
					\end{minipage}\\
					Élément fini de Hermite d'ordre 1
				\end{minipage}
				\item $\Phi_i : p \mapsto D^kp(n_i^K)(v_i^1,\cdots,v_i^k)$\\
				Hermite ordre $k$.
			\end{enumerate}
		\end{ex}

		u = u_D & \text{ sur }\Gamma_D \hspace{.5cm} \text{ condition de Dirichlet}
	\end{cases}$
	
	avec $\dfrac{\partial u}{\partial \mathcal V_\mathcal{A}}(\gamma) = \displaystyle\sum_{\alpha,\beta=1}^2 a_{\alpha,\beta}(\gamma)\dfrac{\partial u}{\partial x_\alpha}(\gamma)\mathcal{V}_\beta(\gamma)$
	
	\begin{cor}[de la formule de Green]$ $\\
		\textbf{Si} $\Omega \subset \mathcal C^{0,1}$, 
		\[
		\begin{aligned}
			a_{\alpha\beta} \in H^1(\Omega)\\
			u \in H^2(\Omega) & & v \in H^1(\Omega)
		\end{aligned}
		\]
		
		\textbf{Alors} on a :
		\[
		\displaystyle \int_\Omega \sum_{\alpha,\beta=1}^2 \dfrac{\partial}{\partial x_\beta}\left(a_{\alpha\beta}\dfrac{\partial u}{\partial x_\alpha}\right)(x) v(x) dx = -\int_\Omega \sum_{\alpha,\beta= 1}^2 a_{\alpha\beta} \dfrac{\partial u}{\partial x_\alpha}(x)\dfrac{\partial v}{\partial x_\beta}(x) dx + \int \frac{\partial u}{\partial \mathcal V_{\mathcal A}}(\gamma)v(\gamma)d\gamma
		\]
	\end{cor}

	\subsection{Stockage de la matrice}
	
	On a obtenu : $A_{IJ} = \displaystyle\int_{\Omega_h} \sum_{\alpha,\beta}a_{\alpha\beta}\dfrac{\partial w_J}{\partial x_\alpha}\dfrac{\partial w_J}{\partial x_\beta} + \int a_{00}w_Jw_I$\\
	
	Cette matrice est : $\begin{aligned}[t]
		\to & \text{symérique}\\
		\to & \text{définie positive si $(a_{\alpha\beta}),a_{00})$ tel que la forme bil. associée es coercive}\\
		\to & \text{\underline{creuse}}
	\end{aligned}$\\
	
	Des stockages creux :
	\begin{enumerate}[label=$\star$]
		\item stockage morse désordonné (accessible au moment de l'assemblage)\\
		
		\rouge{$M_h\times M_h$}\\
		
		\rouge{prise en compte des conditions de Dirichlet}\\
		
		
		\item stockage morse ordonné (pratique pour les opérations linéaire, non accessible au moment de l'assemblage)\\
		
		\rouge{$N_h\times N_h$}
	\end{enumerate}

	\subsubsection{Stockage morse ordonnée}
	
	Tel que défini dans MELINA++, librairie Éléments Finis développée à l'IRMAR.\\
	\rouge{\textbf{!} On appelle élément "non nul" $A_{IJ}$ de $AH$ élément tel que $\mathrm{supp} w_J \cap \mathrm{supp} w_I \not=\emptyset$ \textbf{!}}
	
	Ce stockage permet de faire des opérations algébriques sur les matrices après le stockage !
	
	\begin{defi}[Stockage morse ordonnée]$ $\\
		\begin{minipage}{18cm}
			C'est un ensemble de tableaux :
	\begin{enumerate}[label=$\star$]
		\item  $\mathrm{Diag}(l)$, $l\in[\![1,N_h]\!]$ = valeurs des éléments diagonaux de $A_h$
		
		\item $LMat(l)$, $l\in[\![1,N_{L_h}+1]\!]$ : valeurs des éléments "non nuls" de $L_h$.\\
		avec \begin{minipage}[t]{15cm}
			$L_h$ = partie triangulaire inférieure stricte de $A_h$\\
			$LMat(l)$ = $l^{ieme}$ élément de $L_h$ avec $L_h$ ordonnée ligne par ligne.
		\end{minipage}\\
		Pour $l=N_{L_h}+1$, il s'agit d'un élément fictif considéré comme le seul élément de la ligne $N_h$ de $L_h$ (c'est-à-dire $N_h+1$ de $A_h$)
		
		\item $ColInd(l)$, $l\in[\![1,N_h]\!]$ : indice colonne de l'élément en position $l$ dans $LMat$.
		
		\item $FirstAdLi(i)$, $i\in [\![1,N_h]\!]$ : \begin{minipage}[t]{10cm}
			position dans $LMat$ du premier élément stocké de la ligne $i$ de $L_h$ (ligne $i+1$ de $A_h$)
		\end{minipage}\\
		Si il n'y a pas d'élément "non nul" dans la ligne $i$ de $L_h$, on pose : \\
		\hspace*{2.5cm}$FirstAdLi(i) = FirstAdLi(i+1)$
	\end{enumerate}
		\end{minipage}
	\end{defi}
	
	\begin{rmq}
		\begin{minipage}[t]{18cm}
			$FirstAdLi(i+1)-FirstAdLi(i) = $ nbre d'éléments "non nuls"
		\end{minipage}\\
		C'est l'intérêt de l'élément fictif introduit dans la définition !
	\end{rmq}
