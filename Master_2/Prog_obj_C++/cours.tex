	\subsection{Notion de classe en C++}
	
	Définition de base d'une classe :
	
	class MaClasse$\Big\{$\\
	\hspace*{4cm}\begin{minipage}[t]{.3\textwidth}
		private :\\
		\hspace*{.2cm} \begin{minipage}{.7\textwidth}
			// donnees membres\\
			int donnee 1;\\
			int donnee 2;
		\end{minipage}
	\end{minipage}\begin{minipage}[t]{.5\textwidth}
	public :\\
	\hspace*{.2cm}\begin{minipage}{.7\textwidth}
		//constructeurs (optionnels).\\
		MaClasse(...);\\
		//fonctions membres ou méthode\\
		int fonc1();\\
		double fonc2(int);
	\end{minipage}
	\end{minipage}
	\hspace*{4cm}$\Big\};$\\
	
	Les fonctions membres peuvent être définies ailleurs.\\
	MaClasse::MaClasse($\ldots$)\{$\ldots$\}\\
	int MaClasse::fonc1()\{$\ldots$\}\\
	double MaClasse::fonc2(int i)\{$\ldots$\}
	
	\noindent\underline{Usage :} dans un programme appelant.\\
	\hspace*{1.5cm}\begin{minipage}{15cm}
		MaClasse obj1; \hspace{1cm} cout $<<$obj1\\
		MaClasse* obj1\_p; \hspace{1cm} cout $<<$obj
	\end{minipage}
