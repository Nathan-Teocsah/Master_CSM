		$\forall x_1,\ldots,x_p \in \R^n$, $\forall (\lambda_1,\ldots,\lambda_p)\in(\R^+)^p$\\

	\begin{prop}[Enveloppe convexe]$ $\\
		Soit $A$ une  partie convexe de $\R^n$. L'enveloppe convexe de $A$ noté $co(A)$ ou $conv(A)$ est l'intersection de tous les convexes contenant $A$.
	\end{prop}	
	
	\begin{rmq}
		Si $C$ est un convexe contenant $A$, il contient $conv(A)$.
	\end{rmq}
	
	\begin{prop}
		L'enveloppe convexe d'un ensemble convexe d'un ensemble $A$ est l'ensemble des combinaisons convexes finis d'éléments de $A$ :
		\[
			co(A) = \left\{\sum_{i=1}^p \lambda_ix_i\Big|p\geq 1, x_i\in A, \lambda_i \geq 0, \sum_{i=1}^p \lambda_i = 1\right\}.
		\]
		En particulier :\\
		\begin{minipage}{18cm}
			\begin{enumerate}
			\item  Si $A = (a_1,\ldots,a_p)$ alors $conv(A) = \left\{\displaystyle\sum_{i=1}^p \lambda_i a_i\Big|\lambda_i \geq 0,\, \sum_{i=1}^p \lambda_i = 1\right\}$.
			
			\item  Si $A_i$, pour $i=1,\ldots,p$, son des convexes  de $\R^n$, alors :
			\[
				conv(A_1\cup A_2\cup\cdots\cup A_p) = \left\{\sum_{i=1}^p \lambda_i x_i\Big| x_i \in A_i, \lambda_i \geq 0, \sum_{i=1}^p \lambda_ix_i = 1\right\}.
			\]
		\end{enumerate}
		\end{minipage}
	\end{prop}
	
	\subsection{Cônes convexe}
	
	\begin{defi}
		\begin{minipage}[t]{15cm}
			Une partie $\mathcal C$ de $\R^n$ est un cône de sommet $x_0$ si $\forall x \in \mathcal C$ et $\lambda > 0$, on a :
		\end{minipage}
		\[
			x_0 + \lambda(x-x_0) \in \mathcal C.
		\]
	\end{defi}
	
	\begin{rmq}
		\begin{minipage}[t]{15cm}
			\begin{enumerate}[label=$\textbullet$]
				\item  Si $\mathcal C$ est une partie convexe, alors $\mathcal C$ est dit cône convexe.
				
				\item \underline{si $x_0=0$}, alors $\mathcal C$ est un cône si $\lambda\mathcal C \subset \mathcal C$.
			\end{enumerate}
		\end{minipage}
	\end{rmq}
	
	\begin{prop}
		L'intersection d'une famille de cônes convexes de même sommet $x_0$ est un cône convexe.
	\end{prop}
	
	\begin{defi}
		Soit $A \subset \R^n$, l'ensemble :
		\[cone(A) = \left\{\displaystyle\sum_{i=1}^p \lambda_ix_i\Big|\, p \geq 1, x_i \in A, \lambda_i \geq 0\right\},\]
		est dit cône convexe engendré par $A$.
	\end{defi}
	
	\begin{rmq}
		Si $A$ est un convexe, alors $conv(A) = \bigcup_{\lambda\geq 0} \lambda A$.
	\end{rmq}
	
	\begin{defi}
		\begin{minipage}[t]{15cm}
			Soit $C \not= \emptyset$ un convexe fermé de $\R^n$. Un cône asymptote $C_\infty$ est donné par :
		\end{minipage}
		\[
			C_\infty = \left\{d\in \R^n \Big|\, C+d \R^+ \subset C\right\},
		\]
		$d$ = direction asymptotique.
	\end{defi}
	
	\begin{rmq}
		$C_\infty = \left\{d \in \R^n\Big|\, \exists(x_k) \in C^\N, \exists (t_k) \in \R^\N, t_k \to + \infty, \dfrac{x_k}{t_k} \to d\right\}$
	\end{rmq}

	Soit $f : \R^n \to \R\cup\{+\infty\}$. La fonction $f$ est convexe si : $\forall x,y \in \mathrm{dom}(f)$, $\forall \theta \in [0,\,1]$, 
	\[f(\theta x+(1-\theta)y) \leq \theta f(x)+(1-\theta)f(x).\]
	
	\begin{lm}
		[inégalité de Jensen]$ $\\
		\begin{minipage}{18cm}
			Si $f : \R^n \to \R \cup\{+\infty\}$ une fonction convexe, $(x_i)_{i \in [\![1,p]\!]}\in \mathrm{dom}(f)^p$ et $(\alpha_i)_i \in  \left(\R^+\right)^p$ avec $\displaystyle\sum_{i=1}^p\alpha_i = 1$. Alors :
			\[
				f\left(\sum_{i=1}^p\alpha_i x_i\right) \leq \sum_{i=1}^p \alpha_i f(x_i).
			\]
		\end{minipage}
	\end{lm}
	
	\begin{thm}
		Soit $f : \R^n \to \R \cup\{+\infty\}$. 
		
		\begin{minipage}{15cm}
			La fonction $f$ est convexe, si et seulement si, son épigraphe \textcolor{blue}{(strict)}\\ $\mathrm{epi}(f) = \left\{(x,t)\in \mathrm{dom}(f)\times\R\Big|\,\underset{\textcolor{blue}{(>)}}{\geq} f(x)\right\}$ est convexe.
		\end{minipage}
	\end{thm}
	\begin{dem}
		$(\implies)$ Si $f$ est convexe.\
		
		\begin{minipage}{17cm}
		Soient $(x,t),(y,u)$ deux éléments de l'épigraphe de $f$ et $\theta \in [0,1]$. On a alors :
		\begin{equation}
			f(\theta x + (1-\theta) y) \leq \theta f(x) + (1-\theta)f(y) \overset{epi}{\leq} \theta t + (1-\theta) u.
		\end{equation}
		D'où $\left(\theta x + (1-\theta)y,\theta t+(1-\theta) u\right) \in \mathrm{epi}(f)$.\\
		
		$(\Longleftarrow)$ Si $\mathrm{epi}(f)$ est convexe. Soit $x,y \in \R^n$ et $\theta \in [0,1]$. On a alors :
		\[
			(x,f(x)) \in \mathrm{epi}(f),
		\]
		et,
		\[
			(y,f(y)) \in \mathrm{epi}(f).
		\]
		Il vient alors :
		\[
			(\theta x + (1-\theta)y,\theta f(x) + (1-\theta) f(y) ) \in \mathrm{epi}(f),
		\]
		et ainsi : $f(\theta x + (1-\theta) y) \leq \theta f(x) + (1-\theta) f(y)$.
		\end{minipage}
	\end{dem}
	
	\begin{prop}
		\begin{minipage}[t]{15cm}
			Soit $f : K \longrightarrow \R$ avec $K$ convexe, une fonction convexe.
		\end{minipage}
		
		On peut étendre la fonction $f$ à $\R^n$ en une fonction convexe $\tilde{f}$ par  :
		\[
			\tilde f(x) = \begin{cases}
				f(x) & \text{ si } x \in K \\
				+\infty & \text{ sinon}
			\end{cases}.
		\]
	\end{prop}

			\chi_A(x) = \begin{cases}

		\end{minipage}

			\tilde f = f + \chi_K
		\end{equation}
	\end{rmq}

	\subsection{Cas des fonctions différentiables.}
	
	\begin{thm}
		Soit $f : \R^n \to \R$ une fonction différentiable.
		
		\begin{minipage}{17cm}
			Les propositions suivane sont équivalentes : 
			\begin{enumerate}
				\item  $f$ est convexe
				\item  $\forall (x,y) \in \R^n \times \R^n$, $f(y) \geq f(x) + \left<\nabla f(x),y-x\right>$
				\item $\forall (x,y) \in \R^n\times\R^n$, $\left<\nabla f(x) - \nabla f(y), x-y\right> \geq 0$
			\end{enumerate}
		\end{minipage}
	\end{thm}
	
	\begin{thm}[preuve en exercice]$ $
		
		\begin{minipage}{17cm}
			Soit $f : \R^n \to \R$ de classe $C^2$. La fonction  $f$ est convexe, si et seulement si, pour tout $x \in \R^n$, $H(f)(x) \geq 0$ (ie. la matrice hessienne $H(f)$ de $f$ est semi-défini positive pour tous points de $\R^n$).
		\end{minipage}
	\end{thm}
	
	\begin{ex}
		$ $
		
		\begin{minipage}{17cm}
			\begin{enumerate}[label=\textbullet]
				\item Cas linéaire\\
				$f(x) = \underset{i = 1,\ldots,p}\max\left(\left< a_i,x\right>+b_i\right)$ est convexe.
				
				\item Soit $C$ un convexe :
				\begin{enumerate}[label=-]
					\item $S_c(x) = \underset{y \in C}\sup \left< y,x\right>$ est convexe (support d'une fonction).
					\item $f(x) = \underset{y\in C}\sup \norme{x-y}$ est convexe.
				\end{enumerate}
				
				\item Soit $A$ une matrice symétrique. La fonction $\lambda_{\max}(A) = \underset{\norme{x} = 1}{\sup} Y^t A Y$ est convexe.
			\end{enumerate}
		\end{minipage}
	\end{ex}
	
	\begin{defi}
		\begin{minipage}[t]{15cm}
			\begin{enumerate}[label=\textbullet]
				\item  $f$ est quasi-concave sur $\R^n$, si :
				\begin{equation}
					f(\theta x+(1-\theta)y) \geq \min(f(x),f(y))
				\end{equation}
				
				\item  $f$ est quasi-convexe sur $\R^n$ si $\forall x,y\in \R^n$, $\forall \theta \in [0,1]$,
				\begin{equation}
					f(\theta x + (1-\theta) y) \leq \max(f(x),f(y))
				\end{equation}
			\end{enumerate}
		\end{minipage}
	\end{defi}
	
	\begin{rmq}
		\begin{minipage}[t]{15cm}
			Si $f$ est concave (resp. convexe), alors $f$ est quasi-concave (resp. quasi-convexe). (Exercice)
		\end{minipage}
	\end{rmq}
	
	\begin{lm}
		Soient $f,g : \R^n \to \R$ deux fonctions quasi-concaves. Alors :
		
		\begin{minipage}{17cm}
			\begin{enumerate}
				\item $\alpha f$ est quasi-concave, $\forall \alpha \in \R^+$.
				
				\item  $\min(f,g)$ est quasi-concave.
				
				\item Soit $h : \R \to \R$ une foncion croissante, alors $h\circ f$ est quasi-concave.
			\end{enumerate}
		\end{minipage}
	\end{lm}
	
	\begin{thm}
		\begin{minipage}[t]{15cm}
			Soit $f$ une fonction de classe $\mathcal C^1$ sur $\R^n$. La fonction $f$ est quasi-concave, si et seulement si, $\forall x,y \in \R^n$, si $f(y) \geq f(x)$, alors $\left<\nabla f(x),y-x\right> \geq 0$.
		\end{minipage}
	\end{thm}
	
	\begin{prop}[conjugaison]$ $\\
		\[z\in \partial f(x) \iff x \in \partial f^\star(z) \]
	\end{prop}
	\begin{dem}
		$ $\\
		$\begin{aligned}
			z \in \partial f(x) & \iff \forall y & f(y) \geq f(x) +\left<z,y-x\right>\\
			&\iff \forall y & f(y)-\left<z,y\right> \geq f(x) -\left<x,y\right>\\
			&\iff \forall y&  \left<z,y\right> -f(y) \leq \left<z,x\right>-f(x) \\
			&\iff& f^\star(z) =  \left<z,x\right>-f(x) \\
			&\iff& f^{\star\star}(x) = \left<x,z\right>-f^\star(z)\\
			&\iff& x \in \partial f^\star(z)
		\end{aligned}$
	\end{dem}
