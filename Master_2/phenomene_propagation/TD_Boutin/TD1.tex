	\begin{sol}$ $\\
		\begin{enumerate}
			\item Pour $f \in \mathcal C^2(\R_+^\star)$, on vérifie que : $\Delta F(x,y,z) = f_{rr} + \frac2r f_r$, où $F(x,y,z) = f(r)$.\\
			\begin{minipage}[t]{15cm}
				On a : $\partial_x F = \dfrac{\partial r}{\partial x}\dfrac{\partial f}{\partial r} = \dfrac xr f_r$ et ainsi : $\partial_{xx} F = \left(\dfrac1r - \dfrac {x^2}{r^3}\right)f_r + \left(\frac{x}{r}\right)^2f_{rr}$.
			\end{minipage}\\
			Il suite l'équation suivante : $\Delta F = \left(\frac 3r - \frac{r^2}{r^3}\right)f_r + \frac{r^2}{r^2} f_{rr} = f_{rr} + \frac2rf_r$.
		\end{enumerate}
	\end{sol}
