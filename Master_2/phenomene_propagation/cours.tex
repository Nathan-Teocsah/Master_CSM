	\subsection{Discussion}
	
	\textbf{\textcolor{red}{Attention !}} propagation $\not=$ déplacement de matière.\\
	On étudie l'évolution d'une quantité par rapport à un état de référence.
	
	\begin{ex}
		\begin{minipage}[t]{15cm}
			\begin{enumerate}[label=\textbullet]
			\item Corde de guitare, équation des ondes 1D (en première approximation)
			
			\item  radar, one électromagnétiques.
			
			\item sonar, échographie : ondes acoustiques dans un fluide
			
			\item  sismologie, elastodynamie en milieu solide complexe
			
			\item houle, tsunamies : ondes de surface (déformation de surface libre)
			
			\item trafic routier : véhicules en mouvement et leur densité  $\to$ modélisation fluide
		\end{enumerate}
		\end{minipage}
	\end{ex}

	\begin{enumerate}[label=$-$]

		\item réfraction
		
		\item dispersion : les vitesses de propagation peuvent différer d'une solution à l'autre et êre modifiée dans le temps (l'onde se déforme)
		
		\item  amortissement ou amplification de l'onde.
	\end{enumerate}

	\subsection{Modèles}
	
	\subsubsection{Advection d'un polluant}
	
	Concentration $c(t,x)>0$ de polluant dans un champ de vitesse $u(t,x \in \R$, $x \in \R$ la position dans l'espace.\\
	$\to$ \underline{flux d'advection} du polluant $F(t,x) = u(t,x)c(t,x)$.\\
	\begin{figure}[h!]
		\centering
		\includegraphics[scale=.3]{im2.png}
		\caption{Bilan local de \underline{particules}}
	\end{figure}\\
	$c(t+\dt)\dx = c(t)\dx+F(t,x)\dt-F(t,x+\dx)\dt \underset{\text{sol. reg.}}\to \fbox{$\partial_t c + \partial_x F = 0.$}$\\
	
	Si on tient compte d'un éventuel flux de diffusion : \underline{loi de Fick :} $D_{diff} = -D\partial_x c$ ($D$ est une constante de diffusion moléculaire \fbox{$D>0$}).L'équation d'advecion diffusion est alors : 
	\begin{equation}
		\partial_t c + \partial_x(uc) = D\partial_x^2 c,
	\end{equation}
	c'est une équation de conservation (issue d'un bilan local conservatif). En dimension suérieure, la dérivation de l'équation repose sur la formle de Green pour $u(t,x) \in \R^3$ :
	\begin{equation}
		\partial_t c + \mathrm{div}_{\underbrace{(x,y,z)}_{\vec x}} (uc) = \underbrace{\mathrm{div}_{\vec x}\left(D\nabla_{\vec x} c\right)}_{D\text{ indep. de } (\vec x, t) : D\Delta c}
	\end{equation}
	
	\begin{figure}[h!]
		\centering
		\includegraphics[scale=.3]{im3.png}
		\caption{$\Omega$ élément de volume dans $\R^3$.}
	\end{figure} 
	
	\[
		c(t+\mathrm d\vec x)\mathrm{Vol}(\Omega) = c(t,\vec x)\textsc{Vol}(\Omega) - \underbrace{\int_{\partial\Omega} \left(\vec D(t,\vec y)\cdot\vec n (\vec y)\right) \mathrm d\sigma(\vec y)\dt}_{=\int_\Omega \mathrm{div}\vec D\mathrm d\vec x}
	\]
	\[
		\int_\Omega \left[\dfrac{c(t+\dt,\vec x)-c(t,\vec x)}{\dt}+\mathrm{div}D\right]\mathrm d\vec x = 0
	\]
	Il suit :
	\begin{equation}
		\partial_t c + \mathrm{div} D = 0,
	\end{equation}
	(advection "passive" : $u$ ne dépend pas de $c$).
	
	Si $u$ est constante et uniforme, on a : $\underbrace{\partial_t c + u\partial_x c = 0}{\text{eq. de transpor à viesse }u}$  pour le cas 1D sans diffusion.\\
	De façon générale : $\partial_x(uc) = u\partial_x c + c\partial_x u$, donc : $\partial_t c + u\partial_x c = -(\partial_x u) c$ : induit une amplification ou réduction de $c$ le long de certaines courbes.\\
	
	A contrario, si $\partial_x u=0$, c'est une constante le long de certaines courbes.
	
	\subsubsection{Modèles d'équations d'ondes}
	
	$\to$ \underline{Corde endue de masse linéique $\rho$, tension $T$.}
	
	\begin{figure}[h!]
		\centering
		\includegraphics[scale=.3]{im4.png}
		\caption{Illustration d'une corde.}
	\end{figure}
	
	sous hypothèse de petits déplacements et PFD (bilan de forces) :
	\begin{equation}
		\rho \dfrac{\partial^2u}{\partial t^2} = T\dfrac{\partial^2 u}{\partial x^2}
	\end{equation}
	la déformation $u$ se propage le long de la corde.\\
	
	$\to$ \underline{Equation de Maxwell pour l'électromagnétisme}\\
	Champ électrique $\vec E(t,\vec x) \in \R^3$, magnétique $\vec H(t,\vec x) \in \R^3$, induction électrique $\vec D(t,\vec x) \in \R^3$, magnétique $\vec B(t,\vec x) \in \R^3$. En l'absence de charge et courant ("dans le vide")*.\\
	\begin{minipage}[t]{.5\textwidth}
		\textcolor{red}{bilan}$\begin{cases}
			\dfrac{\partial \vec B}{\partial t} = - \vec{\mathrm{rot}}\vec E \\
			\dfrac{\partial\vec D}{\partial t} = \vec{\mathrm{rot}} \vec H
		\end{cases}$\\
		\textcolor{red}{vide "géométrique} $\begin{cases}
			\mathrm{div}\vec D = 0\\
			\mathrm{div} \vec D = 0
		\end{cases}$
	\end{minipage}\begin{minipage}[t]{.5\textwidth}
	\textcolor{red}{loi de comportement.}\\
	$\vec B = \mu \vec H$ : perméabilité magnétique\\
	$\vec D = \varepsilon \vec E$ : permittivité diélectrique du vide.
	\end{minipage}
	
	En combinant les équations (une sorte d'"élimination de Gauss"), il suit : 
	\[\begin{cases}
		\dfrac{\partial \vec H}{\partial t} = -\dfrac1\mu\vec{\mathrm{rot}}\vec E\\
		\varepsilon\dfrac{\partial\vec E}{\partial t} = \vec{\mathrm{rot}} \vec H
	\end{cases}\]
	, ainsi par élimination:
	\begin{equation}
		\varepsilon\dfrac{\partial^2 \vec E}{\partial t^2} + \vec{\mathrm{rot}}\left(\dfrac1\mu\vec{\mathrm{rot}}\vec E\right) = 0.
	\end{equation}
	Si $\varepsilon$ et $\mu$ sont constants, $\vec{\mathrm{rot}}(\vec{\mathrm{rot}} \vec E) = \vec\nabla\left(\underbrace{\nabla\cdot\vec E}_{=0}\right) - \vec\Delta\vec E$. On obtient finalement :
	
	\begin{equation}
		\dfrac{\partial^2 \vec E}{\partial t^2} - c^2\vec\delta\vec E = 0,
	\end{equation}
	avec $c = \dfrac1{\sqrt{\varepsilon\mu}}$.		
	\subsubsection{Modèle de trafic routier (LWR)}
	LWR = Lightill, Whitham, Richards.\\
	Route unidimensionelle $x \in \R$, temps $t>0$.\\
	\begin{figure}[h!]
		\centering
		\includegraphics[scale=.5]{im5.png}
		\caption{Illustration d'un trafic routier.}
	\end{figure}
	véhicule en file indienne le long de l'axe (sans dépassements ni jonctions).\\
	modèle fluide : \begin{minipage}[t]{15cm}
		$\rho(tnx)$ densité de véhicule autour de $x$,\\
		 $v(,x)$ vitesse moyenne autour du point $x$,\\
		  flux de véhicule : $q(t,x) = \rho(,x)v(t,x)$.
	\end{minipage}
	
	\underline{bilan de masse} sur un intervalle $[a;b]$.\\
	$N(t) = \displaystyle\int_a^b\rho(t,x)\dx$\\
	$\dfrac{d}{\dt}N(t) = q(t,b) - q(t,a) = \displaystyle\int_a^b\partial_x q(t,y) \dy$, d'où $\displaystyle\int_a^b \left(\partial_t\rho + \partial_x q\right)\dy = 0$, ceci pour tout $(a,b)$. Donc :
	\begin{equation}
		\partial_t \rho + \partial_x q = 0.
	\end{equation}
	On se donne une loi de comportement : $v = V(\rho)$ avec $V$ une fonction prescrite.\\
	\underline{typiquement :} $ V(\rho) = V_{\max}\left(1-\dfrac\rho{\rho_{\max}}\right)$. On a :
	\begin{equation}
		\partial_t \rho + \partial_x\left(\underbrace{\rho V(\rho)}_{Q(\rho)}\right) = 0,
	\end{equation}
	équation de transport conservative mais non-linéaire.
	
	
	\begin{figure}[h!]
		\centering
		\includegraphics[scale=.3]{im6.png}
		\caption{Illustrations.}
	\end{figure}
	
	\begin{rmq}
		si $\rho$ est solution régulière :
		\begin{equation}
			\partial_t \rho + Q'(\rho)\partial_x \rho = 0,
		\end{equation}
		on en déduit ainsi l'équation suivante : 
		\[Q'(\rho) = V(\rho) + \rho V'(\rho) = V_{\max}\left(1-\dfrac\rho{\rho_{\max}}\right)-\rho\dfrac{V_{\max}}{\rho_{\max}} = V_{\max} \left(1-2\rho/\rho_{\max}\right)\]
		vitesse de propagation des fluctuations de $\rho$ :
		\begin{enumerate}[label=$ $]
			\item  $> 0$ si : $\rho < \rho_{\max}/2$
			\item $<0$ si $\rho>\rho_{\max}/2$
		\end{enumerate}
		tandis que les véhicules ont toujours la vitesse $V(\rho) >0$.
	\end{rmq}
	
	\section{Équation des ondes 1D à vitesse $C$.}
	
	$\begin{cases}
		\partial_{tt} u - c^2 \partial_{xx} u = 0\\
		u(0,x) = u_0(x) & \text{position initiale}\\
		\partial_t u (0,x) = u_1(x) & \text{vitesse initiale}
	\end{cases}$\\
	approche formelle : $\partial_{tt} -c^2 \partial_{xx}  = (\partial_t+c\partial_x)\circ(\partial_t-c\partial_x)$. Deux transport à vitesse $c$ et $-c$.\\
	\underline{Approche "alg. linéaire"}\\
	$v = \partial_t u$, $w = c\partial_x u$.\\
	$\partial_t v = \partial_{tt} u$, $\partial_x w = c\partial_{xx} u \to \partial_t v-c\partial_x w = 0$\\
	$\partial_x v = \partial_{tx} u$, $\partial_t w = c\partial_{xt} u \to \partial_t w - c \partial_x v = 0$.\\
	
	Ainsi : $\partial_t\begin{bmatrix}
		v \\ w
	\end{bmatrix} + A\partial_x\begin{bmatrix}
	v \\ w
	\end{bmatrix} = 0$, avec $A = \begin{bmatrix}
	0 & - c \\-c & 0
	\end{bmatrix} \in \mathcal S_2(\R)$ diagonalisable sur $\R$.\\
	$A = PDP^{-1}$, $D = \begin{bmatrix}
		c & 0 \\0 & -c
	\end{bmatrix}$, $P = \begin{bmatrix}
	1 & 1 \\
	-1 & 1
	\end{bmatrix}$, $P ^{-1} = \dfrac12\begin{bmatrix}
	1 & -1\\
	1 & 1
	\end{bmatrix}$.\\
	
	Posons $\varphi = \begin{bmatrix}
		\varphi_+\\
		\varphi_-
	\end{bmatrix} = P^{-1}\begin{bmatrix}
	v \\
	w
\end{bmatrix}$ \hspace{2cm} $\left(\begin{aligned}
\vec{r_+} = \begin{bmatrix}
	1 \\
	-1
\end{bmatrix} & \text{par} & \lambda_+ = c \\
\vec{r_-} = \begin{bmatrix}
	1\\
	1
\end{bmatrix} & \text{par} & \lambda_- = -c
\end{aligned}\right)$\\

qui vérifie : \fbox{$\partial_t \varphi + D\partial_x \varphi = 0$} : découplage $\begin{cases}
	\partial_t \varphi_+ + c \partial_x \varphi_+ = 0 \\
	\partial_t \varphi_- - c \partial_x \varphi_- = 0
\end{cases}$\\

$\varphi_+(t,x) = \varphi_+(0,x-ct) $, $\varphi_-(t,x) = \varphi_-(0,x-ct)$.\\
alors $\begin{bmatrix}
	v \\
	w
\end{bmatrix} = P\begin{bmatrix}
\varphi_+\\
\varphi_-
\end{bmatrix} = \varphi_+ \vec{r_+} + \varphi_-\vec{r_-}$, et $w(t,x) = x\partial_x u(t,x) = \varphi_-(0,x+ct)-\varphi_-(0,x-ct) = \dfrac12\left[u_1(x+ct) - u_1(x-ct)\right]+\dfrac{c}2\left[v_0'(x+ct)+v_0'(x-ct)\right]$.\\
Par intégration : $u(t,x) = \dfrac12\left[u_0(x+ct)-u_1(x-ct)\right] + \dfrac1{2c}\displaystyle\int_{x-ct}^{x+ct} u_1(y) \dy$.\\
$u(t,x) = \left[\dfrac12 u_0(x+ct)+\dfrac1{2c}\displaystyle\int_0^{x+ct} u_1(y)\dy\right] + \left[\dfrac12u_0(x-ct)-\dfrac1{2c}\displaystyle\int_{x-ct}^{x+ct} u_1(y) \dy\right]$

\section{Systèmes hyperboliques}

\subsection{En dimension 1 d'espace}

$U(t,x) \in \R^n$, $x \in \R$, dimension 1 d'espace.\\
$\begin{cases}
	\partial_t U + A \partial_x U = 0\\
	U(0,x) = U_0(x) \in \R^n, \forall x \in \R
\end{cases}$, avec $A \in \mathcal M_n(\R)$.\\

\begin{defi}
	\begin{minipage}[t]{15cm}
		Un système est dit hyperbolique si $A$ est diagonalisable sur $\R$.
	\end{minipage}
\end{defi}

\begin{prop}
	\begin{minipage}[t]{15cm}
		Le problème es ben posé, ie. admet une unique solution qui dépend "continuement" de $U_0$, si et seulement si, il est hyperbolique. 
	\end{minipage}
\end{prop}

\begin{ex}
	\begin{minipage}[t]{15cm}
		$\star$ $A$ symétrique réelle $ \implies$ EDP hyperbolique (dit symétrique). $A \in \mathcal M_n(\R)$ avec $n$ valeur propre réelles distinctes 2 à 2 (dit hyperbolique stricte).\\
		
		$\star$ $A = \begin{bmatrix}
			0 & 1 \\
			1 & 0
		\end{bmatrix}$, $\mathrm{spec}(A) = \{i,-i\}$, $U = \begin{bmatrix}
		u \\
		v
		\end{bmatrix}\begin{cases}
		\partial_t u -\partial_x v= 0\\
		\partial_t v + \partial_x u = 0
		\end{cases}$\\
		$\implies$ \fbox{$\partial_{tt} u + \partial_{xx} u = 0$} elliptique en $(t,x)$ $ \not =$ pb d'évolution, et mal posé.
	\end{minipage}
\end{ex}
	
	
	Supposons l'EDP hyperbolique.\\
	$A = PDP^{-1}$, $D = \mathrm{diag}(\lambda_1,\ldots,\lambda_n)$, $P = (\vec{r_1},\ldots,\vec{r_n})$.\\
	$P^{-1} = \begin{bmatrix}
		l_1^T\\
		\vdots\\
		l_n^T
	\end{bmatrix}$, $V=P^{-1}U = \begin{bmatrix}
	l_1^TU \\
	\vdots\\
	l_n^T U
	\end{bmatrix} \in \R^n$ vérifie : $\partial_t V+ D \partial_x V = 0$, \begin{minipage}[t]{6cm}
	$ n$ équation de transport scalaires découplées.
	\end{minipage}
	
	On résoud donc : $U(t,x) = \displaystyle \sum_{j=1}^n \underbrace{l_j^T U(0,x-\lambda_j t)}_{scalaire}\vec{r_j}$.
	
	\begin{ex}[physique]$ $\\
		\begin{minipage}[t]{17cm}
			Les équation d'Euler pour l'acoustique fluide.\\
			$\rho(x,t) \ in \R$ densité d'un fluide compressible\\
			$u(x,t) \in \R$ vitesse eulérienne du fluide au point $x$ à l'instant $t$.\\
			Équation de bilan de masse et de quantité de mouvement sur un volume $[a,b]$ en espace :
			\[
				\int_a^b \rho(t+\dt,x)\dx - \int_a^b \rho(t,x)\dx = \left[(\rho u)(t,a)-(\rho u)(t,b)\right]\dt
			\]
			
		\end{minipage}
		
		\begin{minipage}{17cm}
			le fluide avance de $u\dt$ pendant le temps $\dt$.\\
			($\dt \to 0$) \fbox{$\partial_t\rho + \partial_x(\rho u) = 0$} \textcolor{red}{\underline{masse}}\\
			Pour la qdm, les efforts de pressions interviennent dans le bilan :
			\[
			\begin{aligned}
				\int_a^b (\rho u)(t+\dt,x) \dx - \int_a^b (\rho u)(t,x)\dx =& (\rho u)u(t,a)\dt\\
				&-(\rho u)u(t,b)\dt\\
				&+(P(a)-P(b))\dt
			\end{aligned}
			\]
			d'où l'EDP \fbox{$\partial_t (\rho u) + \partial_x(\rho u^2 + P) = 0$}.
		\end{minipage}
	\end{ex}
	
	\subsection{En dimension supérieure d'espace}
	
	$\partial_t U + \displaystyle\sum_{j=1}^d A_j \partial_{x_j} U = 0$ avec $A_j \in \R^{N \times N}$ e $x \in \R^d$ et $U(t,x) \in \R^N$.\\
	Soit $\vec n \in \R^d$ tel que $\norme{\vec n} = 1$.
	\[
			U(t,x) = F(x\cdot \vec n - \lambda t) \vec R
	\]
	$\vec n = $ direction de propagation dans $\R^d$\\
	$\lambda = $ vitesse réelle
	$U$ est solution "onde plane" si  :
	\[
		\left(-\lambda I_n + \displaystyle\sum_{j=1}^d n_j A_j\right)F'(x\cdot \vec n - \lambda t)\vec R = \vec 0.
	\]
	Notons $\vec A(\vec n) = \displaystyle\sum_{j=1}^d n_j A_j$, l'équation revient à avoir : \fbox{$\vec A(\vec n) \vec R = \lambda \vec R$}.\\
	$\lambda :$ vp de $\vec A(\vec n)$ associée au $\vec{\mathrm{vp}}$ $\vec R$.\\
	Dans la direction $\vec n$, $\vec A(\vec n)$ admet au plus $N$ vp qui peuvent dépendre de $\vec n$ (cas d'un milieu anisotrope).\\
	Le problème est dit hyperbolique dans la direction $\vec n \in \mathbb{S}^{d-1}$ si
	\[
		\vec A(\vec n) \text{ est diagonalisable sur } \R.
	\]
	Il est dit hyperbolique s'il l'est dans toutes les directions $\vec n \in \mathbb{S}^{d-1}$.\\
	Si pour tout $\\vec n \in \mathbb{S}^{d-1}$ et $\vec A(\vec n) = P_0 D_0P_0^{-1} \in \mathcal M_n(\R)$, alors des solutions de l'EDP sont de la forme :
	\[
		\sum_{j= 1}^N F_j (x\cdot\vec n - \lambda_j t) \vec R_j,
	\]
	si $\vec A(\vec n) \vec R_j \lambda_j \vec R_j$ pour tout $j$.
	
	\begin{ex}[de l'eq. des ondes anisotropes dans $\R^2$]$ $\\
		\begin{minipage}{17cm}
			$\partial_t^2u-\mathrm{div}_{x,y)}\left(D\nabla_{(x,y)}u\right) = 0$, $D= \begin{bmatrix}
				c_x^2 & 0 \\
				0 & c_y^2
			\end{bmatrix}$ et :
		\end{minipage}
		\begin{minipage}{17cm}
			\[
				\partial_t^2 u - \underbrace{(c_x^2\partial_x^2u+c_y^2\partial_y^2 u)}_{\text{Laplacien diagonal anisotrope}} = 0
			\]
			réécriture sous forme EDP d'ordre 1.\\
			$U = \begin{bmatrix}
				p \\
				q \\
				r
			\end{bmatrix}$, $p = \partial_t u$, $q = c_x \partial_x u$ et $r = c_y \partial_y u$. Ce qui donne : $\begin{cases}
			\partial_t p = \partial_t^2 u\\
			c_x \partial_x q = c_x^2 \partial_x^2 u\\
			c_y\partial_y r = c_y^2 \partial_y^2 u
			\end{cases}$\\
			Ainsi : $\begin{cases}
				\partial_t p - c_x \partial_x q - c_y \partial_y r = 0\\
				\partial_t q = c_x \partial_{tx} u = c_x \partial_x p\\
				\partial_t r = c_y \partial_{ty} u = c_y \partial_y p
			\end{cases}$, d'où :
			\[
				\partial_t U + \underbrace{\begin{bmatrix}
					0 & -c_x & 0 \\
					-c_x & 0 & 0 \\
					0 & 0 & 0
				\end{bmatrix}}_{A_1}\partial_x U + \underbrace{\begin{bmatrix}
				0 & 0 & -c_y \\
				0 & 0 & 0 \\
				-c_y & 0 & 0
			\end{bmatrix}}_{A_2}\partial_y U = 0.
			\]
			$\vec n \in \mathbb{S}^1$, $\vec n = (\cos \theta,\sin\theta)$, $\vec A(\vec n) = A_1\cos\theta+A_2\sin\theta$ avec :
			\[
				\vec A(\vec n) = \begin{bmatrix}
					0 &-c_x \cos\theta & -c_y \sin\theta \\
					-c_x \cos\theta & 0 & 0\\
					-c_y\sin \theta & 0& 0
				\end{bmatrix}
			\]
			$\vec A(\vec n)$ est sym. réelle don diagonalisable sur $\R$.\\
			Pol. caractéristique :
			\[
				-X(X^2-\lambda^2)
			\]
			avec $\lambda = \pm\sqrt{c_x^2 \cos^2\theta+c_y^2 \sin^2\theta} = \lambda(\theta)$ et $\vec n = \begin{bmatrix}
				1 \\
				0
			\end{bmatrix}$
		\end{minipage}
	\end{ex}
	
	\subsection{Relation de dispersion}
	
	\underline{Introduction :} Pour les EDP linéaires à coefficients constants, on peut déduire des solutions sous la forme :
	($d=1$ en espace) $ u(t,x) = ae^{i(kx-\omega t)}$, avec :
	$a$ : amplitude ($a \in \R^N$ si système de taille $N$).\\
	$k$ : nombre d'onde (en espace)\\
	$\omega$ : pulsation en temps.\\
	Si $k \not= 0, \omega \not = 0$ sont réel, on a : $\lambda = \dfrac{2\pi}{\abs{k}}$ longueur d'ondes en espace, $T = 2\pi/\abs{\omega}$ période en temps.\\
	On écrit : $i(kx-\omega t) = ik(x-v_\varphi t)$ avec $v_\varphi = \omega/k$ est ka vitesse de phase = la vitesse de propagation des oscillations en espace.
	
	\begin{figure}[h!]
		\centering
		\includegraphics[scale=.5]{im11.png}
		\caption{Illustration}
	\end{figure}
	On a un lien algébrique à trouver \textcolor{red}{(rel. de dispertion)} entre $k$ et $\omega$, vitesse de phase $\varphi$ peut dépendre de $k$ et $\omega$ et ne pas êre la même pour les modes en espace (de $k$, de $a$ ?).
	
	\subsubsection{Relation de dispersion d'un système hyperbolique}
	
	\begin{equation}\label{eq:hyp}
		\partial_t U + A \partial_x U = 0,
	\end{equation}
	avec $A$ diagonalisable sur $\R$.\\
	$U(t,x) = e^{i(kx-\omega t)}R$, $R \in \R^N \backslash\{0\}$.\\
	solution, si et seulement si, $-i\omega e^{i(kx-\omega t)}R + Aike^{i(kx-\omega t)} = 0$.\\
	$(\omega I + kA) R = 0$ admet une solution no-nulle, si et seulement si, \fbox{$\mathrm{det}(-\omega I+kA) = 0$} \textcolor{red}{relation de dispersion de \eqref{eq:hyp}}.\\
	Si $A$ admet les vp. $(\lambda_j)_{j=1,\ldots,N}$ on obtient alors ici $\omega \in \{k \lambda_j;\;j\in[\![1,N]\!]\}$. On a pour chaque solution $\omega_j=k\lambda_j$ : $v_\varphi = \dfrac{\omega_j}{k} = \lambda_j$, la vitesse caractéristique déjà rencontré. ($\lambda_j$, indép. de $k$ pour tout $j$).
	
	\begin{defi}$ $\\
		\begin{minipage}{17cm}
			L'équation $D(\omega,k) = 0$ rekuabt $\omega$ et $k$ est la relation de dispersion. Lorsqu'elle peut être résoudre sous la forme : $\omega = \omega_j(k)$, on appelle :
			\begin{enumerate}[label=$\textbullet$]
				\item branche de dispersion des fonction $\omega_j$
				\item polarisation du mode $k$ est le vecteur $R_j(k)$ (ici (-$\omega_j I+k A)R_j)$
				\item vitesse de phase du mode $k$ : $v_\varphi = \dfrac{\omega_j(k)}{k}$
			\end{enumerate}
		\end{minipage}
	\end{defi}
	De façon générale, l'EDP scalaire :
	\begin{equation}
		\sum_{j=1}^s \sum_{l=1}^r a_{jl}\partial_t^j \partial_x^l u = 0,
	\end{equation}
	admet une solution $ae^{i(kx-\omega t)}$ non-nulle, si et seulement si : $\partial_t \longrightarrow -\omega$ et $\partial_x \longrightarrow ik$.
	\[
		\sum_{j=1}^s\sum_{l=1}^r a_{je} (-i\omega)^j (ik)^l = 0,
	\]
	c'est la relation de dispersion et $\omega_j(k) , j \in [\![1,s]\!]$ branche de dispersion.
	
	\begin{rmq}
		$ $\\
		\begin{minipage}{17cm}
			Si $\mathrm{Im}(\omega_j(k)) = \alpha \not = 0$, alors $ae^{i(kx-\mathrm{Re}\omega t)}e^{\alpha t}$\\
			explosion si $\alpha >0$ et amortissement si $\alpha <0$ pour le terme de gauche et transport a vitesse $v_\varphi = \mathrm{Re}\omega/k$.
		\end{minipage}
	\end{rmq}
	\begin{rmq}
		$ $\\
		\begin{minipage}{17cm}
			\begin{enumerate}[label=\textbullet]
				\item avec la relation de dispersion, on peut "algébriser la physique de l'équation.
				\item  hiérarchiser des modèles de la physique à partir de développement asymptotique de leur relations ou branches de dispersion.
			\end{enumerate}
		\end{minipage}
	\end{rmq}
	
	\begin{ex}[non-dispersif]$ $\\
		\begin{minipage}{17cm}
			le modèle est dit non-dispersif si $v_\varphi$ est indépendant sur chaque bande du nombre d'onde $k$.\\
			\begin{enumerate}[label=\textbullet]
				\item $\partial_t u + a\partial_x u = 0 \longrightarrow -i\omega + iak = 0 \iff \omega = ak :$ $v_\varphi = \dfrac{\omega}k = a$ (1 seule bande indépendante de $k$).
				
				\item acoustique linéarisée : $\partial_t \begin{bmatrix}
					p\\
					u
				\end{bmatrix}+\begin{bmatrix}
				u_{eq} & \rho_{eq}c^2 \\
				1/\rho_{eq} & u_{eq}
				\end{bmatrix}\partial_x \begin{bmatrix}
				p \\
				u
				\end{bmatrix} = 0$ avec $c = \sqrt{P'(\rho_{eq})}$.
				
				\[
					(-i\omega I + ik(u_{eq} I + A))R = 0, R \in \C^2
				\]
				\fbox{$(-\omega+ku_{eq})I+kA)=0$}, donc $\mathrm{spec}(A) = \{\pm c\}$ et on trouve donc deux branches : $\omega_+(k) = k(u_{eq}+c)$, $u_-(k) = k (u_{eq}-c)$. avec $R_+ = \begin{bmatrix}
					\rho_{eq}\\
					1
				\end{bmatrix}$ et $R_- = \begin{bmatrix}
				-\rho_{eq}c \\
				1
				\end{bmatrix}$.\\
				$v_{\varphi_+} = \dfrac{\omega_+}{k} = u_{eq}+c$ et $v_{\varphi_-} = \dfrac{\omega_-}k = u_{eq}-c$, ainsi $R_+$ et $R_-$ sont indépendants de $k$.
				
				\item ondes 1D : $\partial_t^2 u - c^2 \partial_x^2 u = 0$.\\
				$\rightarrow (-i\omega)^2-c^2(ik)^2 = 0$, et ainsi $-\omega^2 + c^2 k^2 = 0 \iff \omega = \omega_{\pm} (k) = \pm ck$.\\
				On a :$v_{\varphi_+} = \dfrac{ck}{k} = c$ et $v_{\varphi_-} = \dfrac{-ck}k = -c$.
			\end{enumerate}
		\end{minipage}
	\end{ex}
	
	\begin{ex}[Modèle dispersif de Klein-Gordon]\label{ex:klein}$ $\\
		\begin{minipage}{17cm}
			$\partial_t^2 u - c^2 \partial_x^2 u + \alpha^2 u = 0 $ avec $\alpha \not = 0$ et $\alpha^2 u$ est un terme dispersif.\\
		$\omega^2 = c^2k^2+\alpha^2$, $\omega_{\pm=}(k) = \pm\sqrt{c^2k^2+\alpha^2}$ (deux branches).\\
		Pour $\omega_+(k)$, on trouve la vitesse de phase $v_\varphi(k) = \dfrac{\sqrt{c^2k^2+\alpha^2}}{k} = \sqrt{c_0^2 + \dfrac{\alpha^2}{k^2}}$. Ainsi $v_\varphi$ dépend de $k$ $\longrightarrow$ modèle dispersif.
		\[
			v_{\varphi}(k) \underset{k \to + \infty}{\longrightarrow} c_0.
		\]
		\end{minipage}
		\begin{minipage}{17cm}
			\underline{Fréquence de coupure :} fixons $\omega \in \R$, le nombre d'onde $k$ vérifie : $k^2 = \dfrac{\omega^2-\alpha^2}{c^2}$, donc deux situations surviennent :
		\begin{enumerate}[label=\textbullet]
			\item $\abs{\omega} >\abs{\alpha}$, on a deux modes propagatifs $e^{i(kx-\omega t)}$ avec $v_\varphi = \dfrac{\omega}k \in \R$.
			\item $\abs{\omega}  \abs{\alpha}$, $k = \pm i \sqrt{\dfrac{\alpha^2-\omega^2}{c^2}}$, on obtient un mode évanescent en espace $e^{-i\omega t}e^{ikx} = e^{-i\omega t}e^{\mp\sqrt{\frac{\alpha^2-\omega^2}{c^2}}x}$.
		\end{enumerate}
		\end{minipage}
	\end{ex}

	\subsubsection{Apparition de la vitesse de groupe}
	
	Considérons deux modes pour les nombres d'ondes $k_1$ et $k_2$ distincts.\\
	$\begin{aligned}
		u(t,x) &= \cos\left(k_1 x - \omega_1 t\right) + \cos(k_2x-\omega_2 t)\\
		&= 2\cos\left(\dfrac{k_1-k_2}2 x - \dfrac{\omega_1-\omega_2}2 t\right)\cos\left(\dfrac{k_1+k_2}2 x - \dfrac{\omega_1+ \omega_2}2 t\right)
	\end{aligned}$\\
	avec $\omega_1 = \omega_j(k_1)$ et $\omega_2 = _omega_j(k_2)$ avec $j$ fixé.
	
	\begin{figure}[h!]
		\centering
		\includegraphics[scale=.7]{im13.png}
		\caption{Illustration.}
	\end{figure}
	\underline{Viesse de goupe :} $v_g(k)$ est la vitesse de propagation de l'enveloppe de longueur d'onde $\dfrac{4\pi}{k_2-k_1}$ grande.\\
	\underline{Vitesse de phase :} $v_{\varphi}(k)$ est la vitesse de porteuse $\cos(kx)$.\\
	Pour une branche (ou une modèle) non dispersif, $v_\varphi = v_g$.
